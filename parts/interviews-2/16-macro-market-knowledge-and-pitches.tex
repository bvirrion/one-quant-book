\chapter{Macro, Market Knowledge and Pitches}\label{ch:iv2:macro-market-knowledge-and-pitches}

\begin{ivtrains}
How rates, inflation and growth move assets, what an event does to prices once you ask what was priced in, how to keep up with levels, and how to pitch a stock or a trade. Every level here is invented for the arithmetic. Before any interview, replace them with today's, from a page you can name.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{Analyst, Trader, QT} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:macro-market-knowledge-and-pitches:1}
Yields rise 1 point on a surprise. What happens to a bond with a duration of 7? To a stock valued as $D/(r - g)$ with $r = 8\%$ and $g = 4\%$? What if growth expectations also rise by half a point?
\end{question}

\begin{question}[$\star$ \iqroles{Analyst, Trader, QT} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:macro-market-knowledge-and-pitches:2}
A five-year nominal government bond yields 4.2\%, and the inflation-linked one yields 1.7\%. What is the break-even inflation rate? What does it mean, and what else is in it?
\end{question}

\begin{question}[$\star$ \iqroles{Analyst, Trader, QR} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:macro-market-knowledge-and-pitches:3}
An inflation print comes in above consensus. What moves, in which direction, and why? Go through short rates, long rates, the currency, equities and gold.
\end{question}

\begin{question}[$\star$ \iqroles{Analyst, Trader, QT} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:macro-market-knowledge-and-pitches:4}
The policy rate is 4.33\%, a meeting is coming, and the futures for the period after it price 4.20\%. What probability of a 25 basis point cut is priced? The central bank holds. What happens to short rates and equities, and why does ``the central bank held'' tell you nothing by itself?
\end{question}

\begin{question}[$\star$ \iqroles{Analyst, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:macro-market-knowledge-and-pitches:5}
A company reports in dollars, and 80\% of its sales are in Europe. The euro falls 10\% against the dollar, and local sales don't change. What happens to reported revenue? What would you ask next?
\end{question}

\begin{question}[$\star$ \iqroles{Analyst, Trader, QT, QR} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:6}
Your interviewer asks where the ten-year yield, the main equity index, oil and the main currency pairs are. How do you prepare for this, and what do you say if you don't know?
\end{question}

\begin{question}[$\star$ \iqroles{Analyst, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:macro-market-knowledge-and-pitches:7}
An airline burns 1 billion gallons of fuel a year and makes \$1 billion of operating profit. Crude rises \$10 a barrel, and a barrel holds 42 gallons. What is the rough hit to profit? How would half of the fuel being hedged change it?
\end{question}

\begin{question}[$\star\star$ \iqroles{Analyst, Trader, QR} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:8}
The two-year yield is 4.5\%, and the ten-year is 4.0\%, annually compounded. What eight-year rate, starting in two years, does that imply? Does an inverted curve mean a recession is coming? What does it say exactly?
\end{question}

\begin{question}[$\star\star$ \iqroles{Analyst, Trader} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:9}
Earnings estimates rise 5\% on a stronger economy, while the discount rate rises 50 basis points. The stock has a duration of 20. What is the net effect? When do stocks and yields rise together?
\end{question}

\begin{question}[$\star\star$ \iqroles{Analyst, Trader, QR} \iqfirm{bank, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:10}
A policy rule says $i = r^* + \pi + 0.5(\pi - \pi^*) + 0.5y$, with $r^* = 2\%$, $\pi^* = 2\%$, inflation of 3\% and an output gap of 1\%. What rate does it give? What if inflation rises by a point? What can't a rule like this tell you?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, Risk, Analyst} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:11}
A 60/40 portfolio holds equities with 16\% volatility and bonds with 7\%. What is its volatility at a stock-bond correlation of $-0.3$, and of $+0.3$? What makes the sign change?
\end{question}

\begin{question}[$\star\star$ \iqroles{Trader, QT, Analyst} \iqfirm{bank, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:12}
Payrolls beat consensus by 150,000. On past data, two-year yields move 1 basis point per 30,000 of surprise. What is the expected move? What do you lose on \$10,000 of DV01 long? Why might yields do nothing, or even fall?
\end{question}

\begin{question}[$\star\star$ \iqroles{Trader, Analyst, QR} \iqfirm{bank, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:13}
A central bank cuts 50 basis points when 25 was priced. On past data, a surprise passes through at 60\% to two-year yields and 20\% to ten-year yields. What happens to each, and to a \$10,000 DV01 steepener, long the two-year and short the ten-year? What might the rest of the market do?
\end{question}

\begin{question}[$\star\star$ \iqroles{Trader, QT, Analyst} \iqfirm{bank, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:14}
You borrow yen at 0.5\% and invest in dollars at 4.5\%. How far can the dollar fall against the yen in a year before you lose? With a volatility of 10\%, what is the Sharpe? What happens in an unwind?
\end{question}

\begin{question}[$\star\star$ \iqroles{Analyst, Trader, QR} \iqfirm{bank, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:15}
What does a central bank do when it buys or sells bonds on a large scale, and how does that reach the yield? Why is it hard to measure?
\end{question}

\begin{question}[$\star\star$ \iqroles{Analyst, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:macro-market-knowledge-and-pitches:16}
A company beats on earnings by 5\%, but cuts guidance. The stock's P/E falls from 20 to 18. What happens to the price? What did the market care about?
\end{question}

\begin{question}[$\star\star$ \iqroles{Analyst, Trader, QT} \iqfirm{hedge fund, bank} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:17}
Pitch me a stock. What structure do you use, what must be in it, and how do you make your own?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{Analyst, Trader, QR} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:18}
You pitch a stock at \$50. Bull case: EPS of 3.60 at a P/E of 18, with probability 50\%. Base case: 3.30 at 16, 30\%. Bear case: 2.70 at 12, 20\%. What is the expected return? What is the reward to risk? If you plan to stop out at \$44 and may lose 1\% of the portfolio on the idea, what size do you take? What can go wrong with that stop?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{Trader, QT, Analyst} \iqfirm{bank, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:19}
Pitch a DV01-neutral 2s10s steepener: long the two-year, short the ten-year, \$10,000 of DV01 per side. Modified durations are 1.9 and 8. Yields are 4.0\% and 4.2\%, and repo is 4.1\%. What are the notionals? What do you make on 25 basis points of steepening? What is the yearly carry? With a daily spread volatility of 6 basis points, what is the 99\% one-day VaR?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{Analyst, QR, Risk} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:20}
Inflation is 3\% a year. An inflation-linked bond and a nominal bond each have a duration of 8. The nominal yields 4\%, and the linked one yields a real 1.7\%. Compare one-year returns if nominal yields rise 1 point with real yields unchanged, and if real yields rise 1 point with break-evens unchanged. What does an inflation-linked bond hedge, and what doesn't it?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{Analyst, QR, Trader} \iqfirm{bank, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:21}
The ten-year nominal yield rises 50 basis points in a week: real yields rise 60, and break-evens fall 10. What moved, and what's the story? Compare the moves in nominal bonds of duration 7.5 and linked bonds of duration 8. Would you rather own growth or value stocks, and why?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{Analyst, QR, Trader} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:22}
Set out the growth-and-inflation grid: growth up or down, inflation up or down. What do equities, bonds and commodities tend to do in each square, and which assets win? Why is a table like this dangerous?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, Risk, Analyst} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:23}
A US investor holds euro equities, which return 10\% in euros. The euro falls 5\%. Dollar rates are 4.5\% and euro rates 2.5\%. What is the unhedged dollar return? What is the return with a one-year hedge on the starting value? Why is a positive rate difference a gain for this hedge?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{Trader, QR, Analyst} \iqfirm{bank, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:24}
The policy rate is 4.33\%. A meeting falls in the middle of month 1, and the month 1 future averages 4.25\%. The next meeting is on the first day of month 2, and that future averages 4.10\%. A move is 25 basis points. What probabilities of a cut do these imply at each meeting? What do you need to adjust in practice?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{Trader, Analyst, QR, Risk} \iqfirm{hedge fund, bank} \iqlevel{experienced}]\label{iq:iv2:macro-market-knowledge-and-pitches:25}
Inflation data comes tomorrow, with consensus at 0.3\% month on month. Your view: 0.2\% with probability 25\%, 0.3\% at 50\%, 0.4\% at 25\%. You map those to equity returns of $+1\%$, $+0.3\%$ and $-1.5\%$, and two-year yield moves of $-8$, 0 and $+10$ basis points. You hold \$100 million of equities and \$50,000 of DV01 long two-year bonds. Find the P\&L in each state, its mean and its standard deviation. An at-the-money index straddle for the day costs 1\%. Is it worth buying? What do you do tonight, and what would you tell your manager?
\end{question}
