\chapter{Crypto for Traders and Researchers}\label{ch:iv2:crypto-for-traders-and-researchers}

\begin{ivtrains}
Perpetual swaps and funding, inverse and linear contracts, liquidation, automated market makers, arbitrage across venues and stablecoins. Prices here are invented, venues are unnamed, and a box gives dates where a real number appears.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QT, Trader, QR} \iqfirm{crypto} \iqlevel{junior}]\label{iq:iv2:crypto-for-traders-and-researchers:1}
You are long \$1 million of a perpetual swap that charges 0.01\% funding every 8 hours. What do you pay per interval, per day and per year? If the perpetual trades above spot, who pays whom?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, QR} \iqfirm{crypto} \iqlevel{junior}]\label{iq:iv2:crypto-for-traders-and-researchers:2}
Bitcoin is 60\,000 in spot, and the 90-day future is 61\,500. What is the simple annualised basis? What trade captures it, and what does it earn if held to expiry?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, QD} \iqfirm{crypto} \iqlevel{junior}]\label{iq:iv2:crypto-for-traders-and-researchers:3}
An inverse contract is margined in bitcoin. Its notional is in dollars, and its P\&L in bitcoin is $N(1/P_0 - 1/P_1)$. You are long \$1 million at 50\,000. What is your bitcoin P\&L at 60\,000 and at 40\,000? What is the dollar P\&L at those prices?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, Dev} \iqfirm{crypto} \iqlevel{junior}]\label{iq:iv2:crypto-for-traders-and-researchers:4}
You open a long at 50\,000 with 10x leverage, isolated margin, and a maintenance margin of 0.5\% of entry notional. At about what price are you liquidated?
\end{question}

\begin{question}[$\star$ \iqroles{QT, QR, Dev} \iqfirm{crypto} \iqlevel{junior}]\label{iq:iv2:crypto-for-traders-and-researchers:5}
A constant-product pool holds 100 ETH and 200\,000 USDC, with no fee. What does it cost to buy 10 ETH? What is your average price, and what is the pool's marginal price afterwards?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader} \iqfirm{crypto} \iqlevel{junior}]\label{iq:iv2:crypto-for-traders-and-researchers:6}
A dollar stablecoin trades at 0.995, and the issuer redeems at 1.00 with a fee of 0.1\%. What is the arbitrage, and what does it earn per turn? What can stop it from closing the gap? What happens if it trades at 1.01?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, LLE} \iqfirm{crypto} \iqlevel{junior}]\label{iq:iv2:crypto-for-traders-and-researchers:7}
Venue A bids 60\,000, and venue B offers at 59\,940. Taker fees are 5 basis points at each. Is there a profit in buying on B and selling on A?
\end{question}

\begin{question}[$\star$ \iqroles{QT, QR} \iqfirm{crypto} \iqlevel{junior}]\label{iq:iv2:crypto-for-traders-and-researchers:8}
You deposit equal values of ETH and USDC into a constant-product pool. The ETH price then doubles. What is your loss, as a share of what you would have had by just holding? What about if it halves?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, QR} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:9}
You are long \$1 million of spot and short \$1 million of a perpetual, with funding of 0.01\% every 8 hours, and margin of 20\% on the short. What return do you earn on the total capital? What is it at 50\% margin? What ends the trade?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, QR} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:10}
The 90-day future trades 2.5\% above the perpetual, in total over the 90 days and not annualised. Funding is 0.01\% per 8 hours, paid by longs to shorts, and you assume that it holds. What is the trade between the perpetual and the future, and what does it earn on \$1 million? Why doesn't it matter where spot ends up? What risk remains?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, QD, QR} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:11}
You hold 1 bitcoin, at 60\,000, and you short \$60\,000 of the inverse contract. What is your dollar value at 30\,000 and at 90\,000? Now the same person is long the inverse contract instead of short. What is the shape of the bitcoin P\&L, and why does it matter for margin?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Risk, QR} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:12}
Each 1\% fall in price forces \$30 million of long positions to be liquidated, and the book can absorb \$50 million per 1\% of price move. A shock takes 5\% off the price. How far does the price go in the end? What happens if the liquidations are \$50 million per 1\%?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Risk, Trader, Dev} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:13}
A long of 20 BTC at 50\,000 with 10x leverage has a bankruptcy price of 45\,000, and is liquidated at 45\,250. The fill comes at 44\,500. Who bears the shortfall, and how much? And if the fill is at 45\,250? What happens when the insurance fund is empty?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, QR, Dev} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:14}
The pool holds 100 ETH and 200\,000 USDC, with a fee of 0.3\% on the input. What does it cost to buy 10 ETH now? What is the fee? Over what range of outside price is there no arbitrage against the pool as it stands?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, LLE} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:15}
ETH is 2\,000 dollars, BTC is 60\,000 dollars, and ETH/BTC trades at 0.0340. Taker fees are 10 basis points per leg. What is the loop, and what does it earn? What fee per leg would kill it?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, QR, Risk} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:16}
A dollar stablecoin has 8\% of its reserves stuck at a failed bank, and the rest is safe. You think half of the stuck money comes back. What is it worth? It trades at 0.88. What does a price below 0.92 tell you? How do you size this?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, QR, Risk} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:17}
An algorithmic stablecoin lets holders burn 1 stable for \$1 of its sister token at the market price. The sister token trades in a constant-product pool holding 200 million tokens and 200 million dollars. Holders redeem \$100 million and sell the tokens into the pool. How many tokens are minted? What happens to the price? What is the problem with the design?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Dev, QR} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:18}
A venue computes funding as $F = P + \mathrm{clamp}(I - P, -0.05\%, +0.05\%)$, per 8 hours, with an interest term $I = 0.01\%$ and a premium $P$. What is the funding at $P$ of 0.02\%, 0.06\%, 0.10\%, $-0.03\%$ and $-0.20\%$? What shape is this rule?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Risk, Trader} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:19}
You keep 30\% of capital on a venue. You think it has a 5\% chance of failing per year, with a 60\% loss on what you hold there. What is the expected yearly cost? What changes if you split that across three venues with independent failures? Why is a hedged book worse off when a venue fails than the plain loss says?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT} \iqfirm{crypto} \iqlevel{experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:20}
A perpetual pays continuous funding at a rate $\kappa(F - S)/S$ from longs to shorts, the rate is $r$, and the coin pays no yield. What is the no-arbitrage price $F$? Evaluate it for 8-hour funding, $\kappa = 1\,095$, and $r = 5\%$. What is the funding at that price?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QT, Risk, QR} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:21}
You are short a perpetual with margin that liquidates you on a rise of 10\% at 10x, 20\% at 5x and 50\% at 2x. Volatility is 80\% a year and there is no drift. What is the chance of liquidation within 30 days at each level?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QT, Trader, Risk} \iqfirm{crypto} \iqlevel{experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:22}
You run an arbitrage between two venues with \$1 million on each. You earn 10 basis points net of fees on \$5 million a day. Inventory drifts, so you move \$2 million a day between them at 5 basis points. A transfer takes 30 minutes at 80\% volatility. What is your return on capital? What is the risk per \$1 million moved? How do you remove it, and what does that cost?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT} \iqfirm{crypto} \iqlevel{experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:23}
A constant-product pool has a value of $V = 2\sqrt{kP}$. Hedge it continuously and show that the hedged position loses at a rate $\sigma^2/8$ of pool value per unit of time. Evaluate at 80\% volatility. What is the expected loss in a year, allowing for the pool's value decay? Why does it not vanish when the price comes back?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:24}
A pool holds 1\,000 ETH and \$2 million at 2\,000. You short 1\,000 ETH once and do nothing more. What is your P\&L if ETH goes to 2\,420? Compare this with the impermanent loss. Then the price returns to 2\,000. What is your P\&L now? How does this relate to the continuously hedged loss?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Dev} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:25}
A concentrated position has liquidity in the range 1\,800 to 2\,200, with the price at 2\,000. By how much does it multiply capital efficiency, relative to full-range? What is its loss relative to holding, at the two ends of its range, and for full range? What happens beyond the range?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QT, Dev, QR} \iqfirm{crypto} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:26}
A victim buys 10 ETH from a pool of 100 ETH and 200\,000 USDC, with a fee of 0.3\% and a slippage limit of 1\%. An attacker trades before and after in the same block. What is the largest size the attacker can use? What does the attacker make, before transaction costs? Why is that less than the victim's extra cost? What would make this harmless?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{crypto} \iqlevel{experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:27}
A 30-day at-the-money call on bitcoin at 60\,000, with 80\% volatility and zero rates, settles in the coin: payoff $(S - K)^+/S$ bitcoin. What is its price in coin, and in dollars? What is the delta quoted in coin? Why does it differ from the delta of the ordinary dollar option? Check by parity.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{crypto, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:28}
A researcher shows you a funding carry strategy that made 30\% a year on notional in a one-year test, with margin of 20\%, a Sharpe ratio of 10, and no losing month. Entry and exit cost 20 basis points a leg on two legs, and the position is cycled 12 times a year. Give the net return on capital. Add a 5\% yearly venue failure losing half the capital. What is the Sharpe now, with 2\% of other noise? What else is wrong with the test?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QT, Trader, Risk, QR} \iqfirm{crypto} \iqlevel{experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:29}
You are starting a basis desk with \$10 million. You hold spot against a short perpetual at a chosen leverage, and funding averages 10\% a year. The short is liquidated if the price rises by one over the leverage, and you then close at a cost of 1\% of notional. Reset monthly, with a month being 30 days of a 365-day year. At leverage of 2, 5 and 10, what are the gross yield, the liquidation chance per month with 80\% volatility, and the net yield? Which do you pick? What else do you need before you run it?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{crypto, hedge fund} \iqlevel{experienced}]\label{iq:iv2:crypto-for-traders-and-researchers:30}
You are offered a \$10 million ETH/USD constant-product pool at 80\% volatility, and hedge its delta. Arbitrageurs trade at every hour. What do you lose at a fee of zero, and what is the fee income from arbitrageurs alone at 5, 30 and 100 basis points? How much flow from uninformed traders do you need at each to break even? Would you take it? What changes at 40\% volatility, or in a concentrated range?
\end{question}
