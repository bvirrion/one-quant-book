\chapter{Research and Strategy Cases}\label{ch:iv2:research-and-strategy-cases}

\begin{ivtrains}
Signal design, backtest critique, costs and capacity, sizing and ramp-up, what live trading does to a simulation, take-home data cases, and how to size a desk or a venture. Every number is invented, and each one is meant to be reused: replace it with your own and the method still works.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QR, QT, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:1}
A strategy makes 8 basis points a day on average, with a daily standard deviation of 1.2\%. What is its annualised Sharpe ratio, on 252 days?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:2}
A backtest has a Sharpe ratio of 1 over 4 years. What is its $t$-statistic? How long would a Sharpe of 0.5 need for the same $t$-statistic?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT} \iqfirm{hedge fund} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:3}
A book makes 10 basis points a day gross on capital, and trades 40\% of its capital per day at 5 basis points per dollar traded. Daily volatility is 60 basis points. What are the gross and net Sharpes?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:4}
You may trade 5\% of daily volume, and it takes 5 days to build a position. A stock trades \$50 million a day. What is the largest position? With 200 such stocks, what is the largest gross book?
\end{question}

\begin{question}[$\star$ \iqroles{QT, QR, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:5}
A strategy wins 1 unit 55\% of the time, and loses 1 unit otherwise. What is the expected value per trade? What is the Sharpe per trade, and for 100 independent trades a year?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Dev} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:6}
A daily return is standard normal. A researcher goes long if today's return is positive and short if it is negative, and books today's return. What Sharpe does that show? What does it become with the correct one-day shift?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:7}
You test 20 strategies with no skill at a 5\% significance level, one-sided and independent. What is the chance that at least one looks significant? How many are expected to?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:8}
For a strategy with normal annual returns, what is the chance of a losing year at a Sharpe of 1 and of 2?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:9}
Thirty of 1,000 stocks a year go to zero, and the others return 10\% on average. A backtest uses only stocks that are alive today. What does it show, and what is the true figure?
\end{question}

\begin{question}[$\star$ \iqroles{QR, Risk, QT} \iqfirm{any} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:10}
A strategy has 25\% annual volatility. To run at 10\%, what leverage do you use? How does that change its Sharpe ratio and its return?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:11}
You try 100 independent strategies with no skill over 5 years, and pick the best one. What Sharpe ratio do you expect to see? Why is the quick rule $\sqrt{2\ln N}$ a little high? What is a backtest's headline Sharpe of 1 now worth?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:12}
A signal has an information coefficient of 0.03, and cross-sectional stock volatility is 2\% a day. What is the expected top-minus-bottom decile spread per day? Assume the signal and returns are jointly normal.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:13}
A signal follows an AR(1) with a daily coefficient of 0.9. What is its half-life? If it predicts 4 basis points a day today, what total return does it predict? What share of that do you capture by holding for 6.6 days, and for 20? How do you use this?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:14}
You regress 20-day forward returns on a slow signal, using daily observations, 1,000 days. What is wrong with the usual $t$-statistic, and by how much? What if the signal changes completely every day? What do you use instead?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:15}
Impact in basis points is $\sigma\sqrt{q}$, with $\sigma = 200$ basis points a day and $q$ the order as a share of daily volume. What is the impact at 5\% of volume? At 2.5\%? What does halving your order size do to total impact cost, and what does splitting it across two days do?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:16}
A strategy has a gross edge of 30 basis points per unit traded. Cost grows as $10\sqrt{Q}$ basis points, with $Q$ in units of \$100 million. What size maximises profit? What is the profit and the net edge there? At what size is the profit zero?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:17}
A strategy goes live. How long until a true Sharpe of 2, 1 and 0.5 gives a $t$-statistic of 2? What does that imply for ramp-up? Propose a schedule, with the rules that move you up and down.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:18}
A backtest made 20 basis points per dollar traded, and live trading made 11. You attribute 3 to delay, 2 to spread, 2 to impact and 1 to missed fills. What remains? How do you find it? Which of these get worse with size?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, Risk, QT} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:19}
Your estimate of a strategy is a 15\% return and a 10\% volatility. What is the Kelly leverage, and the expected log growth? If the true return is half that, what do you earn at that leverage? And at half-Kelly?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Dev} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:20}
You get a take-home with five years of daily prices for 500 stocks. Before fitting anything, what do you check? Give the order, and how you'd show it in the write-up.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:21}
Two signals have information coefficients of 0.03 and 0.02, and their correlation is 0.5. What is the best combination, and its information coefficient? What if the correlation is 0.667? What does a low-correlation weak signal do?
\end{question}

\begin{question}[$\star\star$ \iqroles{Analyst, QR, Trader} \iqfirm{hedge fund, bank} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:22}
You are asked to size a desk: \$200 million of capital, 10\% volatility, a Sharpe of 1, and fixed costs of \$10 million. What are the P\&L, the net return on capital and the break-even Sharpe? What numbers would change your mind?
\end{question}

\begin{question}[$\star\star$ \iqroles{Analyst, QT, Trader} \iqfirm{crypto, market maker} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:23}
A market-making venture targets a market with \$1 billion of daily volume, a 2\% share and a net capture of 1 basis point. What is the annual revenue? Fixed costs are \$2 million. What share breaks even? What do you do with this?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund} \iqlevel{junior}]\label{iq:iv2:research-and-strategy-cases:24}
A trend strategy wins 35\% of the time, with an average win of 3 and an average loss of 1. What is the expectancy per trade and the standard deviation? What is the Sharpe over 100 independent trades a year? How often will you see 10 losses in a row in a given 10-trade run?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Analyst} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:25}
A strategy returns 15\% at 10\% volatility, with a beta of 0.5 to a market with an 8\% return and 16\% volatility. Take the risk-free rate as zero. What are its alpha, residual volatility and information ratio? Why would you report those and not the Sharpe?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Analyst} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:26}
A candidate shows a backtest with a Sharpe of 3.5 over two years on large caps, 5 basis points of cost, rebalanced at the close. What do you ask? Give the order and what each answer would change.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund} \iqlevel{experienced}]\label{iq:iv2:research-and-strategy-cases:27}
You have 100 strategies with no skill. Split the 5 years in half. Pick the best in the first half. What is its expected first-half Sharpe? What is its expected second-half Sharpe, and its rank? How do you turn that into a test of overfitting?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Analyst} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:28}
You regress noise on 20 random features with 500 points. What $R^2$ do you expect in sample and out of sample? Why does out-of-sample go negative? What does this say about a reported in-sample $R^2$ of 4\%?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:29}
A strategy has a Sharpe of 2 at 10\% volatility in calm periods (80\% of the time), and a Sharpe of $-1$ at 20\% volatility in stressed periods (the other 20\%). What are its mean, volatility and blended Sharpe? Why is that lower than either the average or what a calm-only sample shows? What would you do?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:30}
A spread follows an Ornstein-Uhlenbeck process with a half-life of 10 days and a stationary standard deviation of 2\%. You enter at 2 standard deviations, and exit at 0, with a cost of 0.2\%. What is the profit per trade? About how long does it take? What is the return if you are always in a trade? What does the model leave out?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:31}
A strategy returns 12\% at 10\% volatility, and has a 2\% chance per year of a 40\% loss that its backtest never saw. What are the real mean, volatility and Sharpe? How long must a backtest be to see one such event with 90\% probability?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, LLE} \iqfirm{market maker, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:32}
A signal predicts 2 basis points, with a half-life of 200 milliseconds, and it costs 1 basis point to trade. What is the net edge at 50, 200 and 400 milliseconds of latency? What is the break-even latency? What does cutting latency from 200 to 50 buy you, and what would you check first?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:33}
You add a stop-loss at $-5\%$ to a 20-day trade, on a driftless random walk with a daily volatility of 1\%. What happens to the mean return, to the volatility and to the chance of stopping? When does a stop truly help, and when does it hurt?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Dev} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:34}
You have ten persistent features and a 20-day forward return, with no true relation, over 1,000 days. What does five-fold random cross-validation show? What does cross-validation with blocks show, and with blocks with a 20-day purge? Why?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:35}
A strategy with even payoffs wins 52\% of the time. How many trades until a $t$-statistic of 1.96 is expected? How many for 80\% power? How do your answers change for a 51\% win rate?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:36}
Your strategy made 20\% on \$100 million, and 12\% on \$300 million. Fit a net return of $a - b\sqrt{Q}$. What does it predict at \$500 million? What size is optimal if capital costs 5\%? Why should you look at marginal returns, not average ones?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{Analyst, QR} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:37}
Your book runs \$300 million at 10\% volatility and a Sharpe of 1. A vendor sells a dataset for \$1 million a year. Its seller shows an uncorrelated Sharpe of 0.5. What is it worth to you? What if the true Sharpe is half of that? How would you test it before paying?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund} \iqlevel{experienced}]\label{iq:iv2:research-and-strategy-cases:38}
A strategy backtested at a Sharpe of 2.4 over 3 years, and was the best of 200 independent trials. Live for 6 months, it shows 0.5. Your prior for the true Sharpe of any idea is normal with mean 0 and standard deviation 0.5. Estimate the true Sharpe after the backtest, and after the live period. Is the strategy broken? How do you size it, what would make you kill it, and what do you tell the portfolio manager?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{Analyst, QT, QR} \iqfirm{crypto, market maker} \iqlevel{junior, experienced}]\label{iq:iv2:research-and-strategy-cases:39}
You are asked whether to back a crypto market-making venture in a set of markets with \$20 billion of daily volume on each of 365 days a year. Net capture is 0.5 basis points of your traded volume, fixed costs are \$6 million a year, and inventory is fixed at \$20 million whatever your share. You think venue failure costs 0.9\% of inventory a year, and the capital in that inventory costs 15\% a year. Work out revenue at shares of 1\%, 2\% and 5\%, and break-even shares with and without the capital charge, at this capture and at half of it. What would you ask? What would make you say no?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund} \iqlevel{experienced}]\label{iq:iv2:research-and-strategy-cases:40}
A take-home gives you a signal $s$ for 500 stocks: a unit-variance AR(1), half-life 5 days. Daily expected return per unit of $s$ is 2 basis points, and stock noise is 2\% a day. You hold positions proportional to $s$, with gross exposure of 1, refreshed every $h$ days. It costs 5 basis points per dollar traded. Derive gross alpha, turnover, net return and Sharpe in $h$. Which refresh is best, and what would you write up?
\end{question}
