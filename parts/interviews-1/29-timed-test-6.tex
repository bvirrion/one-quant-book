\chapter{Timed Test 6}\label{ch:iv1:timed-test-6}

\begin{ivtrains}
Take it in one sitting, without notes or a calculator.
\end{ivtrains}

\ivtestsection{A}{6}{40}

\begin{question}\label{iq:iv1:timed-test-6:1}
A rook and a king stand on two different random squares of an 8 by 8 chessboard. What is the probability that the rook attacks the king, ignoring any blocking? What is the probability that the king attacks the rook, meaning that they stand on neighbouring squares, including diagonals?
\end{question}

\begin{question}\label{iq:iv1:timed-test-6:2}
Four teams play a round robin: each pair meets once, and each game is a fair coin flip with no draws. What is the probability that some team wins all its games? What is the probability that some team wins all and some team loses all?
\end{question}

\begin{question}\label{iq:iv1:timed-test-6:3}
A 52-card deck is shuffled. What is the expected position of the first ace? What is the probability that the first two cards have the same rank?
\end{question}

\begin{question}\label{iq:iv1:timed-test-6:4}
A stock is at 50, and in one period it will go to 65 or to 45. The interest rate is zero. What is a call with strike 50 worth? What position in the stock and cash replicates it? Does the value change if the stock goes up with a real probability of 90\%?
\end{question}

\begin{question}\label{iq:iv1:timed-test-6:5}
A shuffled deck holds 26 red cards and 26 black cards. What is the expected number of neighbouring cards of the same colour? Separately, what is the expected number of pairs of the same rank in a hand of 5 cards from a standard 52-card deck?
\end{question}

\begin{question}\label{iq:iv1:timed-test-6:6}
Each square of a 3 by 3 grid is coloured black or white with equal probability. What is the expected number of lines (rows, columns and the two diagonals) that have a single colour? How likely is it that one or more of the three rows have a single colour?
\end{question}

\ivtestsection{B}{6}{60}

\begin{question}\label{iq:iv1:timed-test-6:7}
Ten balls are thrown independently and at random into 10 bins. What is the expected number of empty bins? What is the probability that exactly one bin is empty? You may use $0.9^{10} \approx 0.349$ and $9! = 362\,880$.
\end{question}

\begin{question}\label{iq:iv1:timed-test-6:8}
In the dice game of craps, you roll two dice. A total of 7 or 11 wins at once, and a total of 2, 3 or 12 loses at once. Any other total becomes your point, and you then roll until the point appears again (you win) or a 7 appears (you lose). What is the probability of winning?
\end{question}

\begin{question}\label{iq:iv1:timed-test-6:9}
You hold 10 assets in equal weights. Each has a volatility of 20\%, and every pair has a correlation of $0.2$. What is the volatility of the portfolio? What does it tend to as the number of assets grows?
\end{question}

\begin{question}\label{iq:iv1:timed-test-6:10}
A stick of length 1 is cut at a uniform random point. Fix a point $x$ on the stick. What is the expected length of the piece that contains $x$, as a function of $x$? Give it at $x = \frac12$ and $x = \frac14$.
\end{question}

\begin{question}\label{iq:iv1:timed-test-6:11}
A permutation of $\{1, \dots, 10\}$ is chosen at random. What is the distribution of the length of the cycle containing element 1, and what is its mean? What is the probability that elements 1 and 2 lie in the same cycle?
\end{question}

\begin{question}\label{iq:iv1:timed-test-6:12}
You have a fair five-sided die, numbered 1 to 5. Design a way to simulate a fair seven-sided die, and find the expected number of rolls per output. What is the lower bound from information?
\end{question}
