\chapter{Timed Test 3}\label{ch:iv1:timed-test-3}

\begin{ivtrains}
Take it in one sitting, without notes or a calculator.
\end{ivtrains}

\ivtestsection{A}{6}{40}

\begin{question}\label{iq:iv1:timed-test-3:1}
A cereal box holds one of three toys, with probabilities $\frac12$, $\frac13$ and $\frac16$, independently from box to box. On average, how many boxes do you buy to get all three?
\end{question}

\begin{question}\label{iq:iv1:timed-test-3:2}
Three fair dice are rolled. Which is more likely, a total of 9 or a total of 10? Give both probabilities.
\end{question}

\begin{question}\label{iq:iv1:timed-test-3:3}
A source sends four symbols with probabilities $\frac12$, $\frac14$, $\frac18$ and $\frac18$. What is the fewest expected bits per symbol of a prefix code? What is the entropy of the source?
\end{question}

\begin{question}\label{iq:iv1:timed-test-3:4}
$X$ is uniform on $[0, 1]$, and given $X$, $Y$ is uniform on $[0, X]$. What is the probability that $Y > \frac12$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-3:5}
Two players take turns rolling a die, and A goes first. A wins by rolling a 6. B wins by rolling a 5 or a 6. What is the probability that A wins?
\end{question}

\begin{question}\label{iq:iv1:timed-test-3:6}
Three vertices of a regular hexagon are chosen at random. How likely are they to form a right triangle? What about an equilateral triangle?
\end{question}

\ivtestsection{B}{6}{60}

\begin{question}\label{iq:iv1:timed-test-3:7}
A permutation of $1, \dots, 8$ is chosen at random. How likely is it that the element 8 lies in a cycle of length exactly 3? What is the expected number of cycles?
\end{question}

\begin{question}\label{iq:iv1:timed-test-3:8}
A disease affects 1\% of people, independently. You pool the blood of 10 people and test the pool once. If it is positive, you test all 10 individually. What is the expected number of tests per person?
\end{question}

\begin{question}\label{iq:iv1:timed-test-3:9}
You keep rolling a fair die and keep a running total. What is the probability that the running total ever equals exactly 6? What is the long-run chance of hitting a large number $n$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-3:10}
A stick of length 1 is broken at a uniform random point, and the two pieces become the sides of a rectangle. What is the average area of the rectangle? How likely is the area to exceed $\frac{3}{16}$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-3:11}
An at-the-money three-month call on a stock at 100 with 16\% volatility is worth about $0.4S\sigma\sqrt{T}$, with zero rates. Estimate the price of the straddle. What daily move makes a delta-hedged holder break even, using 252 days?
\end{question}

\begin{question}\label{iq:iv1:timed-test-3:12}
A random permutation of $1, \dots, 10$ is written in a line. A position is a local maximum if its value exceeds those of its neighbours, and each end has only one neighbour. How many local maxima are there on average?
\end{question}
