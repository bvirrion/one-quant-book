\chapter{Timed Test 4}\label{ch:iv1:timed-test-4}

\begin{ivtrains}
Take it in one sitting, without notes or a calculator.
\end{ivtrains}

\ivtestsection{A}{6}{40}

\begin{question}\label{iq:iv1:timed-test-4:1}
A drawer holds 5 pairs of different socks. You take out 4 socks at random. What is the probability that no two of them match? What is the probability of exactly one matching pair?
\end{question}

\begin{question}\label{iq:iv1:timed-test-4:2}
$X$, $Y$ and $Z$ are independent exponential variables with mean 1. What is the probability that $X > Y + Z$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-4:3}
Fit a straight line by least squares to the three points $(0, 0)$, $(1, 1)$ and $(2, 1)$. Give the intercept, the slope and the $R^2$.
\end{question}

\begin{question}\label{iq:iv1:timed-test-4:4}
A token starts on square 0 of squares numbered 0 to 6. Each turn you roll a die and move forward by that amount, but if this would go past square 6, you stay where you are. On average, how many rolls does it take to land exactly on 6?
\end{question}

\begin{question}\label{iq:iv1:timed-test-4:5}
Two points are chosen independently and uniformly on a circle of radius 1. What is the expected straight-line distance between them?
\end{question}

\begin{question}\label{iq:iv1:timed-test-4:6}
Three players play rock, paper, scissors, all choosing at random. If all three throw the same or all differ, they replay. Otherwise losers are eliminated, and if two remain they play on. On average, how many rounds does it take to have one winner?
\end{question}

\ivtestsection{B}{6}{60}

\begin{question}\label{iq:iv1:timed-test-4:7}
A fair coin is flipped until either two heads in a row or three tails in a row appear. What is the probability that the two heads come first?
\end{question}

\begin{question}\label{iq:iv1:timed-test-4:8}
You draw independent uniform numbers on $[0, 1]$ and multiply them until the product falls below $\frac12$. What is the expected number of draws?
\end{question}

\begin{question}\label{iq:iv1:timed-test-4:9}
A bet wins with probability $\frac12$ and pays 2 to 1, independently each time. What fraction of wealth should you stake each time to maximise long-run growth? What is the growth rate per bet?
\end{question}

\begin{question}\label{iq:iv1:timed-test-4:10}
Two teams play a best-of-seven series, with independent games. If one team wins each game with probability $0.6$, what is the chance that it wins the series? If the teams are equal, what is the expected number of games?
\end{question}

\begin{question}\label{iq:iv1:timed-test-4:11}
Two fair dice are rolled. Let $M$ be the larger value and $m$ the smaller. What is the covariance of $M$ and $m$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-4:12}
The number of claims in a year is Poisson with mean 10, and each claim's size is independent and exponential with mean 5. Find the mean of the total, and its standard deviation. What is the chance of a total of zero?
\end{question}
