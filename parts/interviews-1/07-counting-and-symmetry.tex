\chapter{Counting and Symmetry}\label{ch:iv1:counting-and-symmetry}

\begin{ivtrains}
Counting without listing: choosing the right objects to count, subtracting what you overcounted, reflecting paths, and noticing
when a symmetry gives the answer with no counting at all. The later questions need a bijection or a group of rotations that you
have to find for yourself.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:counting-and-symmetry:1}
Eight people sit around a round table, and two seatings count as the same if one is a rotation of the other. How many seatings are there? How many keep two particular people from sitting next to each other?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:counting-and-symmetry:2}
How many distinct arrangements are there of the letters of MISSISSIPPI? In how many of them are no two S's next to each other?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, QD} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:counting-and-symmetry:3}
