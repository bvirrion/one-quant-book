\chapter{Waiting Times and Markov Chains}\label{ch:iv1:waiting-times-and-markov-chains}

\begin{ivtrains}
Long-run behaviour and waiting times in random processes: how long until something happens, which of two things happens
first, and where the process spends its time. The later questions need you to find the right states, or a symmetry that removes most
of the algebra.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:1}
You flip a fair coin until two heads in a row appear. What is the expected number of flips? How does that change if you wait instead for a
head followed immediately by a tail?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:2}
You roll a fair die until two sixes come up in a row. What is the expected number of rolls? And for three heads in a row with a fair coin?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:3}
You and a friend take turns rolling a fair die, and you go first. The first to roll a six wins. What is your probability of winning?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:4}
Two players are level, and the first to lead by two points wins. Player A wins each point with probability $0.6$, independently. What is the
probability that A wins, and how many more points are played on average?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:5}
A process has two states. In state A it stays with probability $0.9$ at each step and otherwise moves to B. In state B it stays with
probability $0.7$ and otherwise moves to A. In the long run, what fraction of time is spent in B? How long does a visit to B last on average?
Once the process has settled into its long-run behaviour, what is the correlation between ``in B now'' and ``in B after $n$ steps''?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:6}
A fair coin is flipped until either heads-heads-tails or tails-heads-heads appears as three consecutive flips. You win if heads-heads-tails comes
first. What is your probability of winning?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{market maker, hedge fund, bank} \iqlevel{junior, experienced}]\label{iq:iv1:waiting-times-and-markov-chains:7}
You flip a fair coin until the pattern HTHTH appears. What is the expected number of flips? What is it for HHHTT, and why are the two different when each pattern has the same chance of appearing at any given place?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, QD} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:8}
An ant starts at one corner of a cube. At each step it walks along one of the three edges at that corner, chosen at random. On average, how many
steps does it take to reach the diagonally opposite corner?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:9}
You roll a fair die repeatedly and keep a running total. On average, how many rolls does it take until the total is divisible by 6?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{bank, hedge fund} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:10}
You roll a fair die repeatedly and keep a running total. On average, how many rolls does it take until the total is 10 or more? Give an exact method and a quick
approximation to check it, and then the approximation for a total of 100 or more.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:11}
You repeatedly roll a pair of fair dice and look at their sum. What is the probability that a sum of 12 appears before two consecutive sums of 7?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risk, DS, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:waiting-times-and-markov-chains:12}
Buses leave a stop one after another, and each gap between buses is independently 5 or 15 minutes, with equal probability. You arrive at a random moment. What is your expected wait for the next bus?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risk} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:waiting-times-and-markov-chains:13}
A bacterium dies with probability $1/4$, survives unchanged with probability $1/4$, or splits into two with probability $1/2$. Every bacterium does this independently, once per minute. Starting from one bacterium, what is the probability that the colony eventually dies out?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:waiting-times-and-markov-chains:14}
You keep rolling a fair die and adding the results until your total is 1,000 or more. Which final total is most likely, and what is its probability, to three decimal places?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{market maker, hedge fund, crypto} \iqlevel{junior, experienced}]\label{iq:iv1:waiting-times-and-markov-chains:15}
You and an opponent each name a pattern of three coin flips, such as HTH, and your opponent names first. A fair coin is flipped until one of the two patterns appears, and the owner of that pattern wins. If your opponent names HHH, what should you name, and what is your chance of winning? And if they name HTH? Is there a response that gives you at least some fixed chance, whatever they name?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{market maker, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:waiting-times-and-markov-chains:16}
You stand at the number 12 on a clock face. At each step you move to one of the two neighbouring numbers with equal probability, and you continue until you have visited every number. What is the probability that the 6 is the last new number you visit? On average, how many steps does the whole walk take?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker, bank} \iqlevel{junior, experienced}]\label{iq:iv1:waiting-times-and-markov-chains:17}
A knight starts in a corner of an empty chessboard and moves at random, choosing with equal probability among its legal moves each time. On average, how many moves does it take to return to that corner?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:waiting-times-and-markov-chains:18}
A token sits at a corner of a 3-by-3 grid of squares joined edge to edge, so it has 9 positions and 12 connections between neighbours. At each step it moves to a uniformly chosen neighbouring position. On average, how many steps does it take to reach the opposite corner?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:waiting-times-and-markov-chains:19}
You flip a fair coin until two heads in a row appear. The expected number of flips is 6. What is the variance of the number of flips? How does this compare with the wait for a head followed by a tail?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:waiting-times-and-markov-chains:20}
You shuffle a 52-card deck by repeatedly taking the top card and inserting it at a uniformly random one of the 52 possible positions, which include the top and the bottom. Follow the card that began at the bottom. On average, how many moves does it take until that card has reached the top and has then been inserted? Why is the deck perfectly shuffled at that moment?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:waiting-times-and-markov-chains:21}
Ten coins lie on a table, all showing tails. At each step you pick one coin uniformly at random and turn it over. On average, how many steps does it take to reach all heads? And how many steps does it take to come back to all tails for the first time?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{market maker, hedge fund} \iqlevel{experienced}]\label{iq:iv1:waiting-times-and-markov-chains:22}
You and an opponent each name a pattern of four coin flips, and your opponent names first. A fair coin is flipped until one of the patterns appears, and its owner wins. Your opponent names HTHH. Which pattern gives you the best chance, what is that chance, and what is the expected number of flips before the game ends?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{market maker, hedge fund} \iqlevel{experienced}]\label{iq:iv1:waiting-times-and-markov-chains:23}
You roll a fair die and stop as soon as the last six rolls are all different, that is, when they contain every face once. What is the expected number of rolls?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund, crypto} \iqlevel{experienced}]\label{iq:iv1:waiting-times-and-markov-chains:24}
A cell either dies, with probability $1/3$, or splits into two cells, with probability $2/3$. Every cell behaves independently. You start with one cell. What is the probability that the family eventually dies out? Given that it does, what is the expected total number of cells that ever lived, counting the first?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund} \iqlevel{experienced}]\label{iq:iv1:waiting-times-and-markov-chains:25}
Each individual in a population independently has $k$ children with probability $(1/2)^{k+1}$, for $k = 0, 1, 2, \dots$, and then dies, so one child is the average. You start with one individual in generation 0. What is the probability that generation 10 is empty? Given that it is not empty, what is its expected size? What is the expected generation at which the family dies out?
\end{question}
