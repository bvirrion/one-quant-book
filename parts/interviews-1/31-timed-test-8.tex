\chapter{Timed Test 8}\label{ch:iv1:timed-test-8}

\begin{ivtrains}
Take it in one sitting, without notes or a calculator.
\end{ivtrains}

\ivtestsection{A}{6}{40}

\begin{question}\label{iq:iv1:timed-test-8:1}
Which is more likely: at least one six in 4 rolls of a fair die, or at least one double six in 24 rolls of a pair of fair dice? Give both probabilities. You may use $(\frac{35}{36})^{24} \approx 0.509$.
\end{question}

\begin{question}\label{iq:iv1:timed-test-8:2}
A walker on a square grid starts at the origin and takes 4 steps, each one up, down, left or right with equal probability. How likely is the walker to be back at the origin after the 4 steps?
\end{question}

\begin{question}\label{iq:iv1:timed-test-8:3}
A stock pays a dividend of 4 next year, and the dividend grows at a constant rate forever. If the required return is 8\% and the price is 100, what growth rate is the market pricing in? What is the price if the growth rate is 5\%?
\end{question}

\begin{question}\label{iq:iv1:timed-test-8:4}
You roll three fair dice. What is the probability that the highest and the lowest of the three values differ by exactly 5?
\end{question}

\begin{question}\label{iq:iv1:timed-test-8:5}
$X$ and $Y$ are independent and uniform on $[0, 1]$. What is the probability that $X + Y < 1$, given that $X < \frac12$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-8:6}
A set of 28 dominoes has each unordered pair of the numbers 0 to 6 exactly once, including the doubles. You draw two at random. What is the probability that they share a number, so that they can be placed end to end?
\end{question}

\ivtestsection{B}{6}{60}

\begin{question}\label{iq:iv1:timed-test-8:7}
You draw independent uniform numbers on $[0, 1]$ and stop when a number is smaller than the one before it. What is the expected number of draws?
\end{question}

\begin{question}\label{iq:iv1:timed-test-8:8}
In a match, A needs 2 more wins and B needs 3 more. The match is stopped now, and the stake is split in proportion to each chance of winning, with fair games. What share does A get? What if A wins each game with probability $0.6$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-8:9}
To sample a point uniformly from the unit ball in 3 dimensions, you draw from the enclosing cube and reject the points outside. What is the acceptance rate, and the expected number of draws per accepted point? What is the acceptance rate in 10 dimensions, given that the ball's volume there is $\frac{\pi^5}{120}$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-8:10}
A 52-card deck is shuffled. What is the expected number of kings that appear before the first ace?
\end{question}

\begin{question}\label{iq:iv1:timed-test-8:11}
Five people get into a lift at the ground floor of a building with 6 floors above it. Each chooses a floor independently and uniformly. What is the expected number of stops? What is the probability that all five go to different floors?
\end{question}

\begin{question}\label{iq:iv1:timed-test-8:12}
Two regressors each have correlation $\frac12$ with $y$ and correlation $\frac12$ with each other. What is the $R^2$ of the regression of $y$ on both? What is it with one regressor alone?
\end{question}
