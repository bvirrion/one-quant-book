\chapter{Estimation}\label{ch:iv1:estimation}

\begin{ivtrains}
Approcher le bon ordre de grandeur à un facteur deux ou trois près, sans aucune donnée~: découper la quantité en morceaux que l'on peut encadrer, énoncer
l'intervalle en plus du nombre, et vérifier par une seconde voie. Les questions suivantes portent sur des affirmations relatives aux marchés et aux systèmes, où une bonne estimation permet de montrer qu'une chose ne peut pas être vraie.
Les hypothèses qu'il faut poser relèvent de celui qui les pose~; tout nombre donné dans l'énoncé est exact.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QT, Trader, Analyste, QD, Risque} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:estimation:1}
À peu près combien de secondes y a-t-il dans une année~? Combien d'entre elles tombent pendant les heures de cotation, pour un marché ouvert $6.5$ heures par jour et 252 jours par an~?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, Analyste} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:estimation:2}
Une piscine mesure 50 mètres de long, 25 mètres de large et 2 mètres de profondeur. Combien de temps faudrait-il à peu près pour la remplir avec un seul tuyau d'arrosage~? Donnez un intervalle en plus du nombre.
\end{question}

\begin{question}[$\star$ \iqroles{QD, Dev, LLE} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:estimation:3}
Une bourse numérote ses ordres à l'aide d'un compteur et en reçoit un million par seconde, sans interruption. Au bout de combien de temps un compteur non signé de 32~bits sature-t-il~? Et un compteur de 64~bits~? Si le compteur est remis à zéro chaque jour et que le marché est ouvert $6.5$ heures, combien de bits lui faut-il~?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, Analyste, Risque} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:estimation:4}
On vous affirme qu'une action varie d'environ 5~\% sur une journée typique et qu'elle a une volatilité de 20~\% par an. Ici, la volatilité désigne l'écart type des rendements, et il y a 252 jours de cotation dans une année. Ces deux affirmations peuvent-elles être vraies à la fois~?
\end{question}

\begin{question}[$\star$ \iqroles{QT, QD, Dev, LLE} \iqfirm{teneur de marché, hedge fund} \iqlevel{junior}]\label{iq:iv1:estimation:5}
Deux villes sont distantes de 1\,200~km. La lumière se propage à environ 300\,000~km par seconde dans l'air et à peu près aux deux tiers de cette vitesse dans une fibre de verre. Quel est le délai de propagation à sens unique le long d'une fibre rectiligne, et que pourrait économiser une liaison radio rectiligne~? Quelle distance parcourt un signal dans la fibre en une nanoseconde~?
\end{question}

\begin{question}[$\star$ \iqroles{QD, Dev, LLE, Analyste} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:estimation:6}
Un serveur consomme 500~watts toute l'année. L'électricité coûte 10~centimes le kilowattheure, et le refroidissement ainsi que les autres charges du bâtiment ajoutent 50~\% à la puissance consommée. Combien coûte une année d'électricité~? Si le serveur coûte 10\,000~\$ et dure quatre ans, quelle part de son coût annuel représente l'électricité~?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, Analyste} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:estimation:7}
Environ 200\,000~tonnes d'or ont été extraites depuis le début de l'histoire. L'or a une masse volumique de $19.3$~grammes par centimètre cube, et une once troy vaut $31.1$~grammes. Si la totalité de cet or était coulée en un seul cube, quelle serait la longueur de son arête~? À 3\,000~\$ l'once, combien vaut ce cube~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyste} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:estimation:8}
Estimez le nombre de coiffeurs et de barbiers qui travaillent dans une ville d'un million d'habitants. Donnez un chiffre central, un intervalle, et une vérification qui ne repose pas sur le même raisonnement.
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyste} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:estimation:9}
Un bus scolaire mesure environ 12~mètres de long, $2.5$~mètres de large et 2~mètres de haut à l'intérieur, et une balle de golf a un diamètre d'environ $4.3$~centimètres. Combien de balles de golf entreraient dans le bus~? Obtenez votre réponse par deux voies.
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyste} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:estimation:10}
Estimez combien de personnes dans le monde se trouvent en ce moment à bord d'un vol commercial. Utilisez une estimation fondée sur les passagers et une autre fondée sur les avions, et dites de combien elles diffèrent.
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyste} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:estimation:11}
Estimez le nombre de stations-service dans un pays de 300~millions d'habitants, où la plupart des adultes conduisent. Vérifiez le résultat du côté de l'offre, en vous demandant si une station type pourrait survivre avec ce que votre nombre implique.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risque, DS} \iqfirm{toutes} \iqlevel{junior, confirmé}]\label{iq:iv1:estimation:12}
Vous estimez une quantité comme le produit de quatre facteurs indépendants, et pour chaque facteur votre intervalle à 90~\% va de la moitié de votre estimation au double de celle-ci. Quel est votre intervalle à 90~\% pour le produit~? Pourquoi la réponse n'est-elle pas «~d'un seizième à seize fois~»~? Dans quel cas le serait-elle~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, QD, LLE} \iqfirm{teneur de marché, hedge fund} \iqlevel{junior}]\label{iq:iv1:estimation:13}
Un marché boursier a une valeur totale de 60\,000 milliards de dollars, et chaque action change de mains environ deux fois par an. Une transaction moyenne vaut 20\,000~\$. Le marché est ouvert $6.5$ heures par jour et 252 jours par an. Combien de transactions ont lieu par seconde~? Si chaque transaction s'accompagne de 30 messages d'ordres, et que les pointes atteignent 20 fois la moyenne, quelle bande passante un message de 40~octets exige-t-il à la pointe~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyste} \iqfirm{teneur de marché} \iqlevel{junior}]\label{iq:iv1:estimation:14}
Un marché échange 10~milliards d'actions par jour sur 252 jours par an. Un teneur de marché se trouve d'un côté de 5~\% de ces actions et conserve, après tous les coûts de transaction, entre $0.05$ et $0.2$ centime par action. Estimez son revenu annuel de négociation, avec un intervalle. La fourchette sur la plupart de ces actions est de 1 centime~: que dit votre chiffre central de la part de la demi-fourchette qu'il conserve~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QD, Dev, LLE, DS} \iqfirm{teneur de marché, hedge fund} \iqlevel{junior}]\label{iq:iv1:estimation:15}
Vous stockez chaque mise à jour de cotation pour 5\,000 actions, avec en moyenne 10 mises à jour par seconde et par action sur une journée de $6.5$ heures, et ce sur 252 jours par an. Chaque mise à jour fait 40~octets, et la compression réduit les données d'un facteur 5. Quel espace de stockage une année demande-t-elle~? Combien de temps faut-il pour lire une journée non compressée à un gigaoctet par seconde~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QD, Dev, DS} \iqfirm{hedge fund, banque} \iqlevel{junior}]\label{iq:iv1:estimation:16}
Un backtest lit des barres d'une minute pour 5\,000 actions sur 20 ans, avec 252 jours par an et 390 minutes par jour. Chaque barre fait 40~octets. Vous pouvez lire les données sur disque à un gigaoctet par seconde, et le traitement d'une barre prend 1~microseconde sur un cœur. Combien de temps dure une exécution sur un cœur~? Et sur 16 cœurs~? Qu'est-ce qui vous limite alors~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyste, Risque} \iqfirm{teneur de marché, hedge fund, banque} \iqlevel{junior}]\label{iq:iv1:estimation:17}
Chaque transaction a un côté qui traverse la fourchette et un côté qui reste au carnet. Un marché boursier échange 500~milliards de dollars par jour, sur 252 jours par an, avec une fourchette affichée (quoted spread) moyenne de 2~points de base (centièmes de pour cent). Estimez ce que les investisseurs paient en fourchette chaque année. Que représente cela face à une valeur totale de marché de 60\,000 milliards de dollars~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyste, Risque} \iqfirm{hedge fund, teneur de marché} \iqlevel{junior}]\label{iq:iv1:estimation:18}
Un candidat affirme que sa stratégie a rapporté 1~million de dollars par jour en négociant une seule action. Cette action s'échange pour 100~millions de dollars par jour et a une volatilité journalière de 2~\%. Sans autre détail, est-ce plausible~? Montrez le chiffre qui tranche.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{toutes} \iqlevel{junior, confirmé}]\label{iq:iv1:estimation:19}
Estimez combien de molécules du dernier souffle de Jules César se trouvent dans l'air que vous venez d'inspirer. Supposez qu'une inspiration fait un demi-litre, qu'une mole de gaz occupe 24~litres dans les conditions ambiantes, que le nombre d'Avogadro vaut $6 \times 10^{23}$, que l'air a une masse molaire de 29~grammes, que la pression au niveau du sol vaut $10^5$~pascals, que la pesanteur vaut $10\ \mathrm{m/s^2}$, que le rayon de la Terre vaut 6\,400~km, et que ces molécules se sont depuis réparties uniformément dans l'atmosphère. Quelle est la probabilité que votre inspiration en contienne au moins une~?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader, QD} \iqfirm{toutes} \iqlevel{junior, confirmé}]\label{iq:iv1:estimation:20}
De tête, estimez de combien de façons on peut ordonner un jeu de 52~cartes. Comparez ce nombre à celui des atomes de la Terre, dont la masse vaut $6 \times 10^{24}$~kg, en prenant 30~grammes par mole d'atomes et le nombre d'Avogadro égal à $6 \times 10^{23}$. Si une machine avait vérifié $10^9$ ordres par seconde depuis le Big Bang, soit $4.4 \times 10^{17}$ secondes plus tôt, combien en aurait-elle vu~? Combien de mélanges aléatoires faut-il pour avoir une chance sur deux qu'au moins deux d'entre eux coïncident~?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader, Risque, DS} \iqfirm{hedge fund, teneur de marché} \iqlevel{junior, confirmé}]\label{iq:iv1:estimation:21}
Vous donnez des intervalles à 90~\% pour dix grandeurs sans lien entre elles, et seuls cinq contiennent la vérité. Si vos intervalles étaient vraiment à 90~\%, quelle est la probabilité d'un score de cinq ou moins~? En supposant vos erreurs normales en logarithme, par quel facteur faut-il élargir vos intervalles~? Dites ce que doit devenir un intervalle «~à un facteur 3 près~»~?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risque} \iqfirm{banque, hedge fund} \iqlevel{junior, confirmé}]\label{iq:iv1:estimation:22}
Un indice a une volatilité journalière de 1~\% et des rendements de moyenne nulle. Si les rendements étaient normaux, quelle est la plus forte variation journalière que l'on attendrait sur 100~ans de 252 jours de cotation~? Que donne le même calcul si les rendements suivent une loi de Student $t$ à 3 degrés de liberté, ajustée à la même volatilité~? Que concluez-vous d'une chute journalière d'environ 20~\%~?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader, Analyste} \iqfirm{hedge fund} \iqlevel{junior, confirmé}]\label{iq:iv1:estimation:23}
Une stratégie long-short sur actions négocie 1\,000 actions, chacune avec une valeur échangée journalière de 50~millions de dollars, et vous ne dépasserez pas 5~\% de la valeur journalière de chaque action. Chaque position est tenue 5~jours, et le portefeuille est maintenu plein, de sorte que chaque jour vous clôturez un cinquième du portefeuille et en ouvrez un nouveau cinquième. Quelle peut être la taille du portefeuille, en comptant les longs et les shorts~? Si chaque dollar négocié rapporte 5~points de base après coûts, que rapporte la stratégie sur une année de 252 jours, et quel rendement cela représente-t-il sur le portefeuille~?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QD, Dev, LLE, QT} \iqfirm{teneur de marché, hedge fund} \iqlevel{junior, confirmé}]\label{iq:iv1:estimation:24}
Un programme traite les messages de données de marché un à un, dans l'ordre d'arrivée, et chacun prend exactement $0.8$~microseconde. Les messages arrivent au hasard, selon un processus de Poisson, à raison d'un million par seconde. Estimez le temps d'attente moyen dans la file avant le début du traitement. Maintenant, une rafale arrive à un rythme constant de $1.5$~million par seconde pendant 1~milliseconde, puis les arrivées retombent à un rythme constant de $0.5$~million par seconde. Quelle est la longueur de la file à la fin de la rafale, et au bout de combien de temps est-elle vide~?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT, Trader, Risque} \iqfirm{hedge fund, teneur de marché} \iqlevel{confirmé}]\label{iq:iv1:estimation:25}
Un fonds prétend rapporter 30~\% par an sur un portefeuille de 20~milliards de dollars d'actions américaines, en comptant les longs et les shorts, tout en renouvelant intégralement le portefeuille chaque jour, de sorte qu'il clôt chaque jour 20~milliards de dollars et en ouvre 20~milliards. Le marché échange 500~milliards de dollars par jour et la volatilité journalière vaut $1.5~\%$. Supposez que le coût par dollar négocié vaut $\sigma\sqrt{q}$, où $\sigma$ est la volatilité journalière et $q$ la part de la valeur échangée de la journée qui revient au fonds. Il y a 252 jours par an. L'affirmation est-elle crédible~? Si le signal rapporte 10~points de base par dollar négocié avant coûts, quelle taille de portefeuille maximise le profit, et que rapporte-t-il alors~?
\end{question}
