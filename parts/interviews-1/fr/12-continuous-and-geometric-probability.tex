\chapter{Probabilités continues et géométriques}\label{ch:iv1:continuous-and-geometric-probability}

\begin{ivtrains}
Transformer une question sur des variables uniformes ou normales en une région dont on peut voir l'aire. Les premières questions sont des figures sur un carré~; les suivantes utilisent la symétrie sur les cercles et les sphères, l'angle d'un couple de normales et la structure des espacements.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyste} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:1}
Un bâton de longueur 1 est coupé en un point aléatoire uniforme. Quelle est la longueur moyenne du morceau le plus long~? Quel est le rapport moyen entre le morceau le plus court et le plus long~?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:2}
$X$ et $Y$ sont des variables indépendantes et uniformes sur $[0, 1]$. Quelle est la probabilité que $XY < 1/2$~?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, DS} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:3}
$X$ et $Y$ sont des variables indépendantes et uniformes sur $[0, 1]$. Quelles sont la moyenne et la variance de $|X - Y|$~?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyste} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:4}
Deux personnes conviennent de se retrouver dans un café entre midi et 13 h. Chacune arrive à un instant uniforme et indépendant, choisi dans cette heure, et attend l'autre pendant 15 minutes avant de partir. Quelle est la probabilité qu'elles se retrouvent~?
\end{question}

\begin{question}[$\star$ \iqroles{QR, DS, QD, QT} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:5}
Un point est uniformément réparti dans un cube unité de dimension $d$. Quelle est la probabilité qu'il se trouve à moins de $0.05$ du bord, pour $d = 1$, 10 et 100~? Quelle est la probabilité qu'il se trouve dans la plus grande boule inscrite dans le cube, pour $d = 3$ et pour $d = 10$~? Que dit cela des données de grande dimension~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, Analyste} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:6}
Un bâton de longueur 1 est cassé en deux points uniformes indépendants. Quelle est la probabilité que les trois morceaux forment un triangle~? Si vous le cassez plutôt une fois en un point uniforme, puis que vous cassez le morceau le plus long en un point uniforme, quelle est-elle~? Et si vous cassez un morceau choisi au hasard, chaque morceau ayant la même chance d'être choisi~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:7}
$X$, $Y$ et $Z$ sont des variables indépendantes et uniformes sur $[0, 1]$. Quelle est la probabilité que trois bâtons de ces longueurs puissent former un triangle~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, QD} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:8}
Un plancher comporte des droites parallèles espacées d'une unité. Vous laissez tomber une aiguille de longueur 1 dans une position et une orientation aléatoires. Quelle est la probabilité qu'elle croise une droite~? Si l'aiguille est de longueur 3, combien de croisements fait-elle en moyenne~? Trouvez la seconde sans faire aucune intégrale pour cette longueur.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:9}
Vous placez $n$ points de façon indépendante et uniforme sur un cercle. Quelle est la probabilité qu'ils se trouvent tous dans un même demi-cercle~? Donnez la réponse pour $n = 3$ et $n = 5$. Que dit ce résultat sur la question de savoir si un triangle formé par trois de ces points contient le centre~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:10}
Vous placez 4 points uniformes indépendants sur un cercle, reliez les deux premiers par une corde et les deux derniers par une autre. Quelle est la probabilité que les cordes se croisent~? Si 20 points de ce type sont reliés par 10 paires fixées à l'avance, combien y a-t-il de croisements en moyenne~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:11}
Une corde d'un cercle unité est choisie «~au hasard~». Quelle est la probabilité qu'elle soit plus longue que le côté du triangle équilatéral inscrit dans ce cercle~? Calculez-la pour trois façons naturelles de choisir la corde, et dites ce que montrent les réponses différentes.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, DS} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:12}
$X$ et $Y$ sont des variables normales centrées réduites indépendantes. Quelle est la probabilité que $X > 0$ et $Y > \sqrt{3}\,X$~? Quelle est la probabilité que $X + Y > 0$, sachant que $X > 0$~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS, Risque} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:13}
$X$ et $Y$ sont des variables normales centrées réduites indépendantes. Quelle est la probabilité que $X^2 + Y^2 > 4$~? Quelles sont la moyenne et la variance du plus grand de $X$ et $Y$~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risque, DS, QD} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:14}
Trois machines tombent en panne après des durées exponentielles indépendantes, de taux 1, 2 et 3 par an. Quelle est la probabilité que la machine de taux 1 tombe en panne en dernier~? En moyenne, combien de temps s'écoule-t-il jusqu'à ce que les trois soient tombées en panne~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS, QD, LLE} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:15}
Les ordres arrivent selon un processus de Poisson à raison de 2 par minute. Quelle est la probabilité qu'il arrive au moins un ordre pendant une durée donnée de 30 secondes~? Sachant qu'exactement 3 ordres sont arrivés pendant une minute donnée, quelle est la probabilité qu'aucun ne soit arrivé pendant la première moitié de celle-ci~? En moyenne, à quel instant le premier d'entre eux est-il arrivé~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, DS} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:16}
$X$ et $Y$ sont des variables indépendantes et uniformes sur $[0, 1]$. Quelle est la loi du plus petit d'entre eux divisé par le plus grand~? Quelle est la moyenne de $X/Y$, et comment est-elle compatible avec le fait que la médiane de $X/Y$ vaut 1~?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, QD, Dev, DS} \iqfirm{toutes} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:17}
Vous estimez $\pi$ en lançant $n$ points uniformes sur un carré unité et en comptant ceux situés dans le quart de disque de rayon 1 centré sur l'un des coins. Quelle est l'erreur type de votre estimation~? Environ combien de points faut-il pour être à $0.001$ près avec un niveau de confiance de 95~\%~?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risque} \iqfirm{hedge fund, banque} \iqlevel{junior, confirmé}]\label{iq:iv1:continuous-and-geometric-probability:18}
$X$, $Y$ et $Z$ sont conjointement normales, de moyenne 0, de variance 1 et de corrélation deux à deux égale à $0.5$. Quelle est la probabilité que les trois soient positives~? Donnez une formule en fonction de la corrélation $\rho$. Que se passe-t-il en $\rho = -1/2$, et pourquoi cette valeur est-elle la plus basse possible~?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, teneur de marché} \iqlevel{junior, confirmé}]\label{iq:iv1:continuous-and-geometric-probability:19}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, teneur de marché} \iqlevel{junior, confirmé}]\label{iq:iv1:continuous-and-geometric-probability:20}
Un bâton de longueur 1 est coupé en 4 points uniformes indépendants, ce qui donne 5 morceaux. Quelle est la probabilité que ces morceaux puissent former un polygone à 5 côtés~? Donnez la formule pour $k$ morceaux, et dites pourquoi elle ressemble à la probabilité que $k$ points d'un cercle se trouvent dans un demi-cercle.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, teneur de marché} \iqlevel{junior, confirmé}]\label{iq:iv1:continuous-and-geometric-probability:21}
Trois points uniformes indépendants sur un cercle unité forment un triangle. Quelle est son aire moyenne~?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, teneur de marché} \iqlevel{junior, confirmé}]\label{iq:iv1:continuous-and-geometric-probability:22}
Un point est choisi uniformément dans le carré unité. Quelle est la probabilité qu'il soit plus proche du centre du carré que du bord du carré~?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, teneur de marché} \iqlevel{junior, confirmé}]\label{iq:iv1:continuous-and-geometric-probability:23}
Un très grand nombre d'oiseaux se posent à des positions uniformes et indépendantes le long d'un fil. Chaque oiseau peint la portion de fil comprise entre lui et son plus proche voisin. Quelle fraction du fil est peinte~?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, teneur de marché} \iqlevel{confirmé}]\label{iq:iv1:continuous-and-geometric-probability:24}
Quatre points uniformes indépendants sur une sphère sont les sommets d'un tétraèdre. Quelle est la probabilité que ce tétraèdre contienne le centre de la sphère~? Trouvez une démonstration qui ne nécessite aucune intégrale.
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, teneur de marché} \iqlevel{confirmé}]\label{iq:iv1:continuous-and-geometric-probability:25}

\end{question}
