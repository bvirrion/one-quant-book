\chapter{Timed Test 2}\label{ch:iv1:timed-test-2}

\begin{ivtrains}
Take it in one sitting, without notes or a calculator.
\end{ivtrains}

\ivtestsection{A}{6}{40}

\begin{question}\label{iq:iv1:timed-test-2:1}
An urn holds 5 red and 3 blue balls. You draw without replacement until one colour has run out. On average, how many balls are left in the urn when you stop?
\end{question}

\begin{question}\label{iq:iv1:timed-test-2:2}
You roll a pair of dice repeatedly. On average, how many rolls does it take until you have seen a total of 7 and also a total of 11, in either order?
\end{question}

\begin{question}\label{iq:iv1:timed-test-2:3}
A stock is at 100 and will be at 120 or 90 next period. Interest rates are zero. What is the price of a call struck at 100, and how many shares hedge one call?
\end{question}

\begin{question}\label{iq:iv1:timed-test-2:4}
$X$ and $Y$ are independent and uniform on $[0, 1]$. What is the probability that $\sqrt{X} + \sqrt{Y} < 1$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-2:5}
A fair coin is flipped 20 times and shows exactly 10 heads. How likely is it that exactly 5 of those heads came in the first 10 flips? Give an exact expression and a rough value.
\end{question}

\begin{question}\label{iq:iv1:timed-test-2:6}
A fair coin is flipped until the first head. You are told that the number of flips was even. Given this, what is the expected number of flips?
\end{question}

\ivtestsection{B}{6}{60}

\begin{question}\label{iq:iv1:timed-test-2:7}
A deck of 52 cards is dealt to four players, 13 each. You look at your hand and see exactly two aces. What is the probability that the other two aces are together in one player's hand?
\end{question}

\begin{question}\label{iq:iv1:timed-test-2:8}
Ten couples sit at random around a circular table. On average, how many couples are sitting next to each other?
\end{question}

\begin{question}\label{iq:iv1:timed-test-2:9}
An equally weighted index holds 100 stocks, each with volatility 30\%, and every pair of stocks has the same correlation $\rho$. The index has volatility 15\%. What is $\rho$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-2:10}
$X$ and $Y$ are independent and uniform on $[0, 1]$. What is the probability that $X/Y$, rounded to the nearest integer, is even?
\end{question}

\begin{question}\label{iq:iv1:timed-test-2:11}
In tennis, a player wins each point independently with probability $0.6$. A game is won by reaching 4 points with a lead of at least 2. What is the probability that the player wins the game?
\end{question}

\begin{question}\label{iq:iv1:timed-test-2:12}
Evaluate the sum $\sum_{n \ge 1}n^2/2^n$. What probability model gives the answer directly?
\end{question}
