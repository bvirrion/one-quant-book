\chapter{Speed Drills}\label{ch:iv1:speed-drills}

\begin{ivtrains}
Plain arithmetic against the clock. Prepare a pen and a timer before you turn the page, and take each sheet in one sitting.
\end{ivtrains}
\vfill\clearpage
