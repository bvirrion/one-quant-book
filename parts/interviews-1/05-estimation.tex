\chapter{Estimation}\label{ch:iv1:estimation}

\begin{ivtrains}
Getting within a factor of two or three with no data: break the quantity into pieces you can bound, say the interval as well
as the number, and check it by a second route. The later questions test claims about markets and systems, where a good
estimate shows that something cannot be true. Assumptions you must make are yours; any number the stem gives is exact.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QT, Trader, Analyst, QD, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:estimation:1}
Roughly how many seconds are there in a year? How many of them fall in trading hours, for a market open $6.5$ hours a day on 252 days a year?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:estimation:2}
A swimming pool is 50 metres long, 25 metres wide and 2 metres deep. Roughly how long would it take to fill with a single garden hose? Give an interval as well as a number.
\end{question}

\begin{question}[$\star$ \iqroles{QD, Dev, LLE} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:estimation:3}
An exchange numbers its orders with a counter and receives one million orders per second, without pause. How long until an unsigned 32-bit counter runs out? And a 64-bit one? If the counter resets each day and the market is open $6.5$ hours, how many bits does it need?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, Analyst, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:estimation:4}
Someone tells you a stock moves about 5\% on a typical day and has a volatility of 20\% a year. Volatility here means the standard deviation of returns, and there are 252 trading days in a year. Can both statements be true?
\end{question}

\begin{question}[$\star$ \iqroles{QT, QD, Dev, LLE} \iqfirm{market maker, hedge fund} \iqlevel{junior}]\label{iq:iv1:estimation:5}
Two cities are 1\,200 km apart. Light travels at about 300\,000 km per second in air and at about two-thirds of that in glass fibre. What is the one-way delay along a straight fibre, and what could a straight radio link save? How far does a signal in fibre travel in one nanosecond?
\end{question}

\begin{question}[$\star$ \iqroles{QD, Dev, LLE, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:estimation:6}
A server draws 500 watts all year. Electricity costs 10 cents per kilowatt-hour, and cooling and other overheads in the building add 50\% to the power used. What does a year of power cost? If the server costs \$10\,000 and lasts four years, what share of its yearly cost is power?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:estimation:7}
About 200\,000 tonnes of gold have been mined in all of history. Gold has a density of $19.3$ grams per cubic centimetre, and a troy ounce is $31.1$ grams. If all of it were cast as one cube, how long would its side be? At \$3\,000 an ounce, what is the cube worth?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:estimation:8}
Estimate the number of hairdressers and barbers working in a city of one million people. Give a central figure, an interval, and one check that does not use the same reasoning.
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:estimation:9}
A school bus measures about 12 metres by $2.5$ metres by 2 metres inside, and a golf ball is about $4.3$ centimetres across. How many golf balls would fit in the bus? Reach your answer by two routes.
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:estimation:10}
Estimate how many people in the world are in the air on commercial flights right now. Use one estimate built from passengers and another built from aircraft, and say how far they differ.
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:estimation:11}
Estimate the number of petrol stations in a country of 300 million people, where most adults drive. Check the result from the supply side, by asking whether a typical station could survive on what your number implies.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risk, DS} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:estimation:12}
You estimate a quantity as a product of four independent factors, and for each factor your 90\% interval runs from half your guess to twice your guess. What is your 90\% interval for the product? Why is the answer not ``from one sixteenth to sixteen times''? When would it be?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, QD, LLE} \iqfirm{market maker, hedge fund} \iqlevel{junior}]\label{iq:iv1:estimation:13}
A stock market has a total value of \$60 trillion, and each share changes hands about twice a year. An average trade is worth \$20\,000. The market is open $6.5$ hours a day on 252 days a year. How many trades happen per second? If each trade comes with 30 order messages, and peaks run at 20 times the average, what bandwidth does a 40-byte message need at the peak?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyst} \iqfirm{market maker} \iqlevel{junior}]\label{iq:iv1:estimation:14}
A market trades 10 billion shares a day on 252 days a year. A market maker is on one side of 5\% of those shares and keeps, after all costs of trading, between $0.05$ and $0.2$ cents per share. Estimate its yearly trading revenue, with an interval. The spread on most of these stocks is 1 cent: what does your central figure imply about how much of the half-spread it keeps?
\end{question}

\begin{question}[$\star\star$ \iqroles{QD, Dev, LLE, DS} \iqfirm{market maker, hedge fund} \iqlevel{junior}]\label{iq:iv1:estimation:15}
You store every quote update for 5\,000 stocks, with an average of 10 updates per second per stock over a $6.5$-hour day, on 252 days a year. Each update is 40 bytes, and compression reduces the data by a factor of 5. How much storage does a year need? How long does it take to read one uncompressed day at a gigabyte per second?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QD, Dev, DS} \iqfirm{hedge fund, bank} \iqlevel{junior}]\label{iq:iv1:estimation:16}
A backtest reads one-minute bars for 5\,000 stocks over 20 years, with 252 days a year and 390 minutes a day. Each bar is 40 bytes. You can read data from disk at 1 gigabyte per second, and processing one bar takes 1 microsecond on one core. How long does a run take on one core? And on 16 cores? What limits you then?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyst, Risk} \iqfirm{market maker, hedge fund, bank} \iqlevel{junior}]\label{iq:iv1:estimation:17}
Every trade has one side crossing the spread and one side resting on the book. A stock market trades \$500 billion a day, on 252 days a year, at an average quoted spread of 2 basis points (hundredths of a percent). Estimate what investors pay in spread each year. What is that against a total market value of \$60 trillion?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyst, Risk} \iqfirm{hedge fund, market maker} \iqlevel{junior}]\label{iq:iv1:estimation:18}
A candidate says their strategy made \$1 million a day trading a single stock. That stock trades \$100 million a day and has a daily volatility of 2\%. Without any more detail, is that plausible? Show the figure that decides it.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:estimation:19}
Estimate how many molecules from Julius Caesar's last breath are in the breath you just took. Assume that a breath is half a litre, a mole of gas occupies 24 litres at room conditions, Avogadro's number is $6 \times 10^{23}$, air has a molar mass of 29 grams, surface pressure is $10^5$ pascals, gravity is $10\ \mathrm{m/s^2}$, the Earth's radius is 6\,400 km, and that those molecules have since spread evenly through the atmosphere. What is the chance your breath contains at least one?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader, QD} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:estimation:20}
By hand, estimate the number of orderings of a 52-card deck. Compare it with the number of atoms in the Earth, whose mass is $6 \times 10^{24}$ kg, taking 30 grams per mole of atoms and Avogadro's number as $6 \times 10^{23}$. If a machine checked $10^9$ orderings a second since the Big Bang, $4.4 \times 10^{17}$ seconds ago, how many would it have seen? How many random shuffles give an even chance that two of them match?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader, Risk, DS} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:estimation:21}
You give 90\% intervals for ten unrelated quantities, and only five contain the truth. If your intervals really were 90\% intervals, how unlikely is a score of five or fewer? Treating your errors as normal in logarithms, by what factor should you widen your intervals? Say what an interval of ``within a factor of 3'' should become.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{bank, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:estimation:22}
An index has a daily volatility of 1\% and returns with mean zero. If returns were normal, what is the largest one-day move you would expect in 100 years of 252 trading days? What does the same calculation give if returns follow a Student $t$ distribution with 3 degrees of freedom, scaled to the same volatility? What do you conclude from a one-day fall of about 20\%?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:estimation:23}
A long-short equity strategy trades 1\,000 stocks, each with a daily traded value of \$50 million, and you will not exceed 5\% of any stock's daily value. Each position is held for 5 days, and the book is kept full, so every day you close a fifth of it and open a new fifth. How large can the book be, counting longs plus shorts? If each dollar traded earns 5 basis points after costs, what does the strategy make in a year of 252 days, and what is that as a return on the book?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QD, Dev, LLE, QT} \iqfirm{market maker, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:estimation:24}
A program handles market data messages one at a time, in order of arrival, and each takes exactly $0.8$ microseconds. Messages arrive at random, as a Poisson process, at one million per second. Estimate the average wait in the queue before handling starts. Now a burst arrives at a steady $1.5$ million per second for 1 millisecond, after which arrivals fall to a steady $0.5$ million per second. How long is the queue when the burst ends, and how long until it is empty?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT, Trader, Risk} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:estimation:25}
A fund claims to return 30\% a year on a \$20 billion book of US stocks, counting longs plus shorts, while replacing the whole book every day, so that each day it closes \$20 billion and opens \$20 billion. The market trades \$500 billion a day and daily volatility is $1.5\%$. Assume the cost per dollar traded is $\sigma\sqrt{q}$, where $\sigma$ is daily volatility and $q$ is the fund's share of the day's traded value. There are 252 days in a year. Is the claim believable? If the signal earns 10 basis points per dollar traded before costs, what book size maximises profit, and what does it earn then?
\end{question}
