\chapter{Number Riddles}\label{ch:iv1:number-riddles}

\begin{ivtrains}
Puzzles about whole numbers: the hands of a clock, factorials, remainders, last digits, divisibility, and which amounts can be made from given coins. They reward working modulo a small number and finding a structure rather than a lot of cases. The early questions are warm-ups, and the last ones need real persistence.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:1}
What is the angle between the hour and minute hands of a clock at 4:36? Both hands move continuously.
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, QD, Dev} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:2}
How many zeros does $250!$ end in, when written out in ordinary decimal digits?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:3}
Without a calculator, what is the last digit of $7^{2026}$ when it is written out in decimal?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, QD, Dev} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:4}
A corridor has 1000 closed doors, numbered 1 to 1000. For each $k$ from 1 to 1000, a person walks past and changes the state of every door whose number is a multiple of $k$. How many doors are open at the end? How many times is door 72 changed?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:5}
Take any three-digit number and write it twice to make a six-digit number, as in 123123. Why is the result always divisible by 7, 11 and 13?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:6}
What is the smallest positive whole number that leaves remainder 1 when divided by 2, remainder 2 when divided by 3, and so on up to remainder 9 when divided by 10?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, QD} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:7}
How many zeros does $20!$ end in when it is written in base 12?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:8}
The five-digit number $4a27b$, with digits $a$ and $b$, is divisible by 72. Find $a$ and $b$.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, QD} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:9}
What is the remainder when $3^{2026}$ is divided by 17? Work it out by hand.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:10}
A number leaves remainder 1 when divided by each of 2, 3, 4, 5 and 6, and it is divisible by 7. What is the smallest such positive number, and what is the next one?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, QD} \iqfirm{market maker, hedge fund} \iqlevel{junior}]\label{iq:iv1:number-riddles:11}
You have unlimited coins worth 7 and worth 11. What is the largest amount that you cannot pay exactly? How many positive amounts can you not pay?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, QD, Dev} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:12}
You write out all the whole numbers from 1 to 1000. How many times do you write the digit 1? What is the sum of all the digits of the numbers from 1 to 999?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, QD, Dev} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:13}
What is the smallest set of weights, each a whole number of grams, that lets a balance weigh any object of a whole number of grams from 1 to 40, if a weight may go on either pan? How many weights are needed if they may go on only one pan?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, QD} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:14}

\begin{question}[$\star\star$ \iqroles{QR, QT} \iqfirm{market maker, hedge fund} \iqlevel{junior}]\label{iq:iv1:number-riddles:15}
Without computing it, what is the highest power of 2 that divides $\binom{100}{50}$? What is the highest power of 3? Is $\binom{100}{50}$ divisible by 7?

\begin{question}[$\star\star$ \iqroles{QR, QT} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:number-riddles:16}

\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{market maker, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:number-riddles:18}
Take every repeating decimal $0.abcabcabc\dots$ in which $a$, $b$ and $c$ are three different digits, with zero allowed. What is the sum of all these numbers?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{market maker, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:number-riddles:19}
A clock has hour and minute hands of equal length, so you cannot tell them apart, and both move continuously. During one 12-hour cycle, at how many moments are you unable to tell the time from the positions of the hands?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader, QD} \iqfirm{market maker, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:number-riddles:20}
Boxes hold 6, 9 or 20 items, and you must buy whole boxes. What is the largest number of items that you cannot buy exactly? Prove that every larger number can be bought.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT} \iqfirm{market maker, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:number-riddles:21}
Which positive whole numbers cannot be written as a sum of two or more consecutive positive whole numbers? Prove it. In how many ways can 2026 be written as such a sum?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, QD, Dev} \iqfirm{market maker, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:number-riddles:22}
100 people stand in a circle, numbered 1 to 100 in order. Going round the circle from person 1, person 1 stays, person 2 leaves, person 3 stays, person 4 leaves, and so on, always skipping one remaining person and removing the next, until one person is left. Which number is left?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{market maker, hedge fund} \iqlevel{experienced}]\label{iq:iv1:number-riddles:23}
What is the smallest positive whole number that doubles when its last digit is moved to the front? For example, moving the last digit of 1234 to the front gives 4123. The number may not begin with 0.
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{market maker, hedge fund} \iqlevel{experienced}]\label{iq:iv1:number-riddles:24}
Let $A$ be the sum of the decimal digits of $4444^{4444}$. Let $B$ be the sum of the digits of $A$, and let $C$ be the sum of the digits of $B$. What is $C$? Find it by hand.
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{market maker, hedge fund} \iqlevel{experienced}]\label{iq:iv1:number-riddles:25}
Five sailors gather a pile of coconuts. During the night each wakes once, in turn. Each divides the pile into five equal shares with one coconut over, gives that coconut to a monkey, hides one share and pushes the other four back into a pile. In the morning the remaining coconuts divide exactly into five equal shares. What is the smallest possible number of coconuts in the original pile?
\end{question}
