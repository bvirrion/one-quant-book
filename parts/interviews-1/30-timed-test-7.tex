\chapter{Timed Test 7}\label{ch:iv1:timed-test-7}

\begin{ivtrains}
Take it in one sitting, without notes or a calculator.
\end{ivtrains}

\ivtestsection{A}{6}{40}

\begin{question}\label{iq:iv1:timed-test-7:1}
You keep rolling a fair die. On average, how many rolls does it take to see both a 1 and a 2? What is it until you have seen all three even faces?
\end{question}

\begin{question}\label{iq:iv1:timed-test-7:2}
A coin with probability of heads $0.1$ is flipped 10 times. What is the probability of an even number of heads? You may use $0.8^{10} \approx 0.107$.
\end{question}

\begin{question}\label{iq:iv1:timed-test-7:3}
$X$ and $Y$ each have variance 1, and their correlation is $\frac12$. What is the variance of $X + 2Y$? What is the slope in a regression of $Y$ on $X + Y$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-7:4}
Two fair dice show $X$ and $Y$. What is the expected value of $\max(X - Y, 0)$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-7:5}
Two fair dice are rolled. Given that at least one shows a 6, what is the probability that both do? If instead you know that the first die shows a 6, what is the probability that both do?
\end{question}

\begin{question}\label{iq:iv1:timed-test-7:6}
A 2 by 2 matrix has four entries, each independently 0 or 1 with equal probability. What is the probability that it is invertible over the real numbers?
\end{question}

\ivtestsection{B}{6}{60}

\begin{question}\label{iq:iv1:timed-test-7:7}
A bag holds three coins: one fair, one with two heads, and one with heads probability $\frac34$. You draw one at random and flip it twice, getting two heads. What is the probability that the next flip of the same coin is a head?
\end{question}

\begin{question}\label{iq:iv1:timed-test-7:8}
A light bulb's life is uniform on $[0, 2]$ years. Each bulb is replaced as soon as it fails, and this has gone on for a long time. At a random moment, what is the expected age of the bulb in use? What is the expected total life of that bulb?
\end{question}

\begin{question}\label{iq:iv1:timed-test-7:9}
$Z$ is standard normal. Using $\varphi(1) = 0.242$ and $P(Z > 1) = 0.159$, find $\mathbb{E}[Z \mid Z > 1]$. A loss distribution is normal with mean 0 and volatility 1. Given the 99\% quantile $2.326$ and $\varphi(2.326) = 0.0267$, what is the expected shortfall at 99\%?
\end{question}

\begin{question}\label{iq:iv1:timed-test-7:10}
A permutation of $\{1, 2, 3, 4\}$ is chosen at random. An inversion is a pair of positions $i < j$ whose values are out of order. Find the mean number of inversions, and its variance.
\end{question}

\begin{question}\label{iq:iv1:timed-test-7:11}
$X$ and $Y$ are standard normals with correlation $\frac12$. What is $\mathbb{E}\max(X, Y)$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-7:12}
You roll 100 fair dice. Estimate the probability that the sum is at least 380, given that one die has variance $\frac{35}{12}$ and that $P(Z > 1.73) = 0.042$ for a standard normal $Z$.
\end{question}
