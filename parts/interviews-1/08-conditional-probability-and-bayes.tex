\chapter{Conditional Probability and Bayes}\label{ch:iv1:conditional-probability-and-bayes}

\begin{ivtrains}
Updating a belief correctly: conditioning on what you actually learned, and on how you came to learn it. Bayes by odds makes most
of these a one-line sum. The traps are protocols of information, base rates, correlated evidence and selection, and the later
questions turn the update into a price or a decision.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:1}
A family has two children, each equally likely to be a boy or a girl, independently. You are told that at least one of them is a girl. What is the probability that both are? And if instead you meet one of the children, picked at random, and that child is a girl?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst, DS, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:2}
A disease affects 1\% of people. A test detects it in 99\% of those who have it, and it wrongly shows positive for 5\% of those who do not. You test positive. What is the probability that you have the disease?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:3}
A prize is behind one of three doors. You pick a door. The host, who knows where the prize is, always opens a different door with no prize behind it, and offers you the chance to switch to the remaining closed door. Should you switch, and what is your chance of winning if you do? What if there are 100 doors and the host opens 98?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:4}
A bag holds 100 coins, of which 99 are fair and one has heads on both sides. You draw one at random and flip 10 heads in a row. How likely is it now that you hold the two-headed coin?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:5}
Three cards lie in a hat: one red on both sides, one blue on both sides, and one red on one side and blue on the other. You draw one at random and put it on the table, and the side you see is red. What is the probability that the other side is red too?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Analyst, DS, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:6}
In a city, 85\% of taxis are green and 15\% are blue. After an accident, a witness says the taxi was blue. Tests show that witnesses identify the colour correctly 80\% of the time, for either colour. What is the probability that the taxi was blue?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:7}
You roll two fair dice, and someone looks at them and tells you that at least one shows a six. What is the probability that both show six? What is the probability that the sum is 7?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:8}
A bag holds one ball, equally likely to be red or blue. You add a red ball, shake the bag and draw one ball, which is red. What is the probability that the ball left in the bag is red?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:9}
Three prisoners, A, B and C, know that one of them, chosen at random, will be pardoned. A asks the guard, who knows the result, to name one of the other two who will not be pardoned. The guard says ``B''. What is A's probability of being pardoned now, and C's, if the guard picks at random when there is a choice? What if the guard always names B whenever possible?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:10}
A prize is behind one of three doors, and you pick door 1. First rule: the host does not know where the prize is, opens door 2 or door 3 at random, and it happens to hide nothing. Should you switch, and how much does it matter? Second rule: the host knows, never opens the prize door, and given a choice always opens door 2. What is your chance if you switch after seeing door 3 opened? After seeing door 2 opened?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:12}
A coin has an unknown probability of heads, and before any flip you think every value between 0 and 1 is equally likely. You see 7 heads in 10 flips. What is the probability that the next flip is heads? Why not $0.7$?

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:13}
A bag holds two coins, one fair and one with heads probability $3/4$. You take one at random, flip it twice, and get two heads. What is the probability that the third flip of the same coin is also heads?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:14}
A condition has a prevalence of 1\%. A test is right 90\% of the time, both on people with the condition and on people without it, and repeated tests are independent given the true state. You test positive twice. What is the probability that you have it? A third test then comes back negative. What now?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS, Risk} \iqfirm{hedge fund, bank} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:16}
You test 1\,000 trading signals, and 5\% of them truly work. A test at the 5\% level flags a useless signal 5\% of the time, and it flags a working signal 80\% of the time. A signal is flagged. What is the probability that it truly works? What should you change first if that is too low?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Analyst, DS, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:19}

What is the probability of guilt?

\begin{question}[$\star\star$ \iqroles{QT, Trader, Analyst} \iqfirm{hedge fund, market maker, bank} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:21}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior}]\label{iq:iv1:conditional-probability-and-bayes:22}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:conditional-probability-and-bayes:36}
An urn is of type A or type B, with equal chance. People guess the type one at a time, and each guess is announced. Before guessing, each person sees a private signal, which names the true type with probability $2/3$, independently of the other signals. Each person also hears all the earlier guesses, but never sees the signals behind them. Each guesses the type that is more likely given all this, and follows their own signal if the two types are equally likely. Eventually, a person's guess no longer depends on their own signal, and from then on everyone guesses the same type. What is the probability that this type is the wrong one? On average, how many people guess before the first person whose signal no longer matters?
\end{question}
