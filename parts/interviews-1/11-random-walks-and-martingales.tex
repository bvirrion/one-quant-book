\chapter{Random Walks and Martingales}\label{ch:iv1:random-walks-and-martingales}

\begin{ivtrains}
Hitting probabilities, durations and extremes of walks, usually without solving any recursion: find what stays fair, stop it, and read off the answer. The later questions ask you to invent the martingale yourself, or to say why optional stopping fails.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:1}
You stand at position 3 on a line numbered 0 to 10. Each step moves you one place left or right with equal probability, and you stop on reaching 0 or 10. What is the probability that you end at 10, and how many steps do you take on average?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:2}
You bet \$1 at a time at even money and win each bet with probability $0.6$. You start with \$5 and stop on reaching \$10 or \$0. What is the probability that you reach \$10, and how many bets does it take on average?
\end{question}

\begin{question}[$\star$ \iqroles{QT, Trader, Analyst, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:3}
You play a fair game, betting \$1 each time at even money, and you stop as soon as you are up \$1 or down \$9. How often do you win the \$1? What is your average profit, and how many bets does a session take on average?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:4}
A walk starts at 0 and moves $+1$ or $-1$ with equal probability at each step. What is the probability that it is back at 0 after 10 steps? How far from 0 is it after 100 steps, on average? Give an exact formula and the quick approximation.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:5}
A walk moves $+1$ with probability $0.6$ and $-1$ with probability $0.4$ at each step. On average, how long does it take to go from 0 to $+10$ for the first time? What identity gives you this without any equations?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:6}
A walk moves $+1$ with probability $0.4$ and $-1$ with probability $0.6$ at each step, forever. What is the probability that it ever reaches $+10$? What is the average of the highest point it ever reaches?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Risk, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:7}
You have \$1,023 and play a fair even-money game by doubling: bet \$1, and after each loss double the bet, stopping as soon as you win or cannot make the next bet. What is the probability you end up \$1 ahead? What is your expected profit? If a win ends the session, how many bets do you make on average?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund, market maker} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:8}
A fair walk starts at 0 and stops the first time it reaches $+1$. It reaches $+1$ with probability one, so its final value is always 1, yet it is a fair game that started at 0. Why does this not contradict the claim that a fair game cannot be beaten by stopping? What is the median stopping time, and how does the mean behave?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, QD} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:9}
A walk on the numbers 0 to 10 moves left or right with equal probability, except that at 0 it always moves to 1. On average, how many steps does it take to go from 0 to 10 for the first time? From 4 to 10? Find a process that stays fair and gives this at once.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:10}
At each step a walk moves $+1$ with probability $0.5$, $-1$ with probability $0.3$, and stays where it is with probability $0.2$. From 0, what is the probability that it reaches $+5$ before $-5$? How long does that take on average?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Risk, QR} \iqfirm{market maker, hedge fund, crypto} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:11}
A price moves one tick up with probability $0.55$ or one tick down with probability $0.45$, independently. You buy, with a profit target 20 ticks up and a stop 10 ticks down. What is the probability that the target is hit first, and how many ticks does the trade take on average? What is your expected profit in ticks? With a fair walk, what would it be?
\end{question}

\begin{question}[$\star\star$ \iqroles{QT, Trader, Risk, Analyst} \iqfirm{hedge fund, market maker, bank} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:12}
You have \$100 and an even-money bet that you win with probability $0.55$, repeated as often as you like. You always bet the same dollar amount. What is the probability of ever going broke if you bet \$1 each time? \$10? \$50?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:13}
A fair walk starts at 0. What is the probability that it stays strictly above 0 for the first 20 steps? What is the probability that it first returns to 0 exactly at step 2? At step 4? At step 6?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{hedge fund, market maker} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:14}
You believe an event has probability $1/3$. Each day brings news that either doubles or halves your odds for the event, and your belief is always the correct conditional probability. What is the probability that your belief ever reaches $8/9$? Why do you need neither the news's reliability nor the number of days?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior}]\label{iq:iv1:random-walks-and-martingales:15}
Two tokens sit on a ring of 12 positions, 4 places apart. At each step both tokens move at the same time, each independently one place clockwise or anticlockwise with equal probability. On average, how many steps until they are on the same position?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:random-walks-and-martingales:16}
A fair walk runs for 20 steps. At which time is its last visit to 0 most likely, and what are the probabilities that it is at time 0, time 10 and time 20? In the long-run limit, what is the probability that the last visit falls in the first quarter of the run?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:random-walks-and-martingales:17}
Three players hold $a$, $b$ and $c$ coins. In each round, one ordered pair of players is chosen uniformly among the six ordered pairs, and the first gives one coin to the second. On average, how many rounds until someone has no coins, starting from 10, 20 and 30 coins? If the game goes on between the two left, with the same rule, how long on average until one player holds all 60?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risk, DS} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:random-walks-and-martingales:18}
An urn starts with one red and one blue ball. Each time you draw one at random, then return it with one more of the same colour. Show that the fraction of red balls is a martingale. Give the best upper bound you can, without simulation, on the probability that this fraction ever reaches $3/4$ or more, and a lower bound from what the fraction does in the long run. What is the true value, roughly?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund, bank} \iqlevel{junior, experienced}]\label{iq:iv1:random-walks-and-martingales:19}
A fair walk runs for 100 steps. What is the average of its highest point, counting the start at 0? Use the reflection principle to reduce it to the distribution of the final value, and compare the result with a continuous approximation.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:random-walks-and-martingales:20}
A walk moves $+1$ with probability $0.6$ and $-1$ with probability $0.4$. The first time it reaches $+10$ takes 50 steps on average. What is the standard deviation of that time? Find it from a martingale, not a recursion.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, QD} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:random-walks-and-martingales:21}
A fair walk moves on the integer lattice, choosing one of the neighbouring points with equal probability at each step, in one, two and three dimensions. In which dimensions does it return to its start with probability one? Give the argument, and the return probability in three dimensions, to two decimal places.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:random-walks-and-martingales:22}
A walk goes up 2 with probability $1/3$ and down 1 with probability $2/3$. From 0, you stop on first being at 10 or more, or at $-5$ or less. What is the probability that you stop at the top, and how many steps does it take on average? Why is one-third, the answer you would get for steps of 1, not quite right?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:random-walks-and-martingales:23}
A fair walk starts at 0 and stops on first reaching $+n$ or $-n$. Its average duration is $n^2$. What is its variance? Give a formula and the value for $n = 10$.
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:random-walks-and-martingales:24}
A fair walk starts at 0 and runs until it first returns to 0. It is certain to return, although the average time is infinite. On average, how many times does it visit the point $+k$ before it returns, for $k = 1$? For $k = 100$? Why is that not absurd?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT, Trader, Risk} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:random-walks-and-martingales:25}
You have \$25 and need \$100. You can stake any whole number of dollars on each round of an even-money bet that you win with probability $0.4$, and you stop on reaching \$100 or \$0. What is your chance if you bet \$1 at a time? What if you bet as much as you can without passing \$100 each time? Is that the best plan? What changes if the win probability is $0.6$ or $0.5$?
\end{question}
