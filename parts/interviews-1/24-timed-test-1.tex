\chapter{Timed Test 1}\label{ch:iv1:timed-test-1}

\begin{ivtrains}
Take it in one sitting, without notes or a calculator.
\end{ivtrains}

\ivtestsection{A}{6}{40}

\begin{question}\label{iq:iv1:timed-test-1:1}
A fair coin is flipped 10 times. What is the probability that no two heads appear in a row?
\end{question}

\begin{question}\label{iq:iv1:timed-test-1:2}
Two players take turns flipping one fair coin, and the first player goes first. A player wins by flipping tails when the flip just before was heads. What is the probability that the first player wins?
\end{question}

\begin{question}\label{iq:iv1:timed-test-1:3}
You have 2 identical eggs and a building of 100 floors. An egg survives a drop from any floor up to some unknown floor, and breaks from any floor above it. A surviving egg can be used again. What is the smallest number of drops that guarantees you find that floor?
\end{question}

\begin{question}\label{iq:iv1:timed-test-1:4}
$X$ and $Y$ are independent and uniform on $[0, 1]$. What is the probability that $\tfrac14 < |X - Y| < \tfrac12$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-1:5}
A strategy has monthly returns that are independent and normal, and its annual Sharpe ratio, mean over standard deviation, is 1. About what fraction of its months lose money?
\end{question}

\begin{question}\label{iq:iv1:timed-test-1:6}
A $5 \times 5$ matrix has 1 in every diagonal entry and $0.25$ in every other entry. What is its determinant?
\end{question}

\ivtestsection{B}{6}{60}

\begin{question}\label{iq:iv1:timed-test-1:7}
A player takes 100 basketball shots, making the first and missing the second. From then on, the probability of making each shot equals the fraction of shots made so far. What is the probability of making exactly 50 of the 100?
\end{question}

\begin{question}\label{iq:iv1:timed-test-1:8}
A stick of length 1 is cut at two independent uniform points. How long, on average, is the longest of the three pieces?
\end{question}

\begin{question}\label{iq:iv1:timed-test-1:9}
Exactly one of 1\,000 wine bottles is poisoned. A test strip dipped in a mix of wines turns positive if the mix contains any of the poison. A positive strip cannot be used again, and a negative one can. You have two rounds of testing: strips of one round are all read together, and you may choose the second round after reading the first. In each round, each strip you use goes into one mix. What is the smallest number of strips you must own to be sure of identifying the bottle?
\end{question}

\begin{question}\label{iq:iv1:timed-test-1:10}
You travel between home and office and own 3 umbrellas. Each trip, it rains with probability $0.3$, independently of the other trips. You carry an umbrella only if it is raining and there is one where you are. In the long run, what fraction of trips leave you wet?
\end{question}

\begin{question}\label{iq:iv1:timed-test-1:11}
Three players fire at one another in the repeating order A, B, C, skipping anyone dead, until one is left. A hits with probability $1/3$, B with probability $2/3$, and C never misses. On a turn a player can aim at either opponent or fire into the air, and each plays to maximise their own chance of surviving. What is the probability that A survives?
\end{question}

\begin{question}\label{iq:iv1:timed-test-1:12}
You test whether a coin is fair, using a two-sided test at the 5\% level. If the coin's probability of heads is in fact $0.6$, you want to detect this with probability 80\%. Using the normal approximation, about how many flips do you need?
\end{question}
