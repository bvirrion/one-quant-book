\chapter{Timed Test 5}\label{ch:iv1:timed-test-5}

\begin{ivtrains}
Take it in one sitting, without notes or a calculator.
\end{ivtrains}

\ivtestsection{A}{6}{40}

\begin{question}\label{iq:iv1:timed-test-5:1}
Events arrive as a Poisson process with rate 2 per hour. What are the mean and variance of the time of the third event? What is the probability that the third event arrives after one hour?
\end{question}

\begin{question}\label{iq:iv1:timed-test-5:2}
You flip a fair coin until the first head, and let $N$ be the number of flips. Then you roll $N$ fair dice and add them up. Find the mean and the variance of the sum.
\end{question}

\begin{question}\label{iq:iv1:timed-test-5:3}
A coin has an unknown probability of heads, with a uniform prior on $[0, 1]$. You see 3 heads in 4 flips. What is the posterior mean? What is the probability that the next two flips are both heads?
\end{question}

\begin{question}\label{iq:iv1:timed-test-5:4}
Five red and five blue balls are lined up in a random order. A run is a maximal block of one colour. What is the expected number of runs? What is the probability of exactly two runs?
\end{question}

\begin{question}\label{iq:iv1:timed-test-5:5}
Three fair dice are rolled. What is the probability that the product of the three numbers is divisible by 6?
\end{question}

\begin{question}\label{iq:iv1:timed-test-5:6}
$X$ and $Y$ are independent standard normals. What is the chance that the point $(X, Y)$ lies within radius 1 of the origin? What radius holds exactly half the probability?
\end{question}

\ivtestsection{B}{6}{60}

\begin{question}\label{iq:iv1:timed-test-5:7}
An election between A and B ends with 7 votes for A and 3 for B, counted in random order. What is the probability that A is strictly ahead throughout the count?
\end{question}

\begin{question}\label{iq:iv1:timed-test-5:8}
In a random graph on 10 vertices, each edge is present independently with probability $\frac12$. Find the mean and the variance of the number of triangles.
\end{question}

\begin{question}\label{iq:iv1:timed-test-5:9}
Using $\log_{10}2 = 0.30103$, how many decimal digits does $2^{1000}$ have, and what is its leading digit?
\end{question}

\begin{question}\label{iq:iv1:timed-test-5:10}
Bus line A arrives every 10 minutes and line B every 15 minutes, each on a fixed timetable, with independent uniform phases. You arrive at a random time and take the first bus. What is your expected wait?
\end{question}

\begin{question}\label{iq:iv1:timed-test-5:11}
A game show has 5 doors, one with a prize. You choose a door. The host, who knows where the prize is, opens one other door with no prize behind it, and offers a switch. If you switch at random to one of the three remaining closed doors, what is your chance of winning?
\end{question}

\begin{question}\label{iq:iv1:timed-test-5:12}
$X$ is binomial with 100 trials and success probability $\frac12$. Bound $P(X \ge 70)$ by Hoeffding's inequality $\exp(-2t^2/n)$, with $t$ the excess over the mean. Compare the bound with Chebyshev's inequality and the normal approximation.
\end{question}
