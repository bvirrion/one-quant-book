\chapter{Expectation and Indicators}\label{ch:iv1:expectation-and-indicators}

\begin{ivtrains}
Computing an average without ever finding the distribution: breaking a count into yes-or-no pieces, summing
tail probabilities, using symmetry, and then going one step further to the variance of the count. The chapter
starts with quick warm-ups and ends with problems that need one real idea you have to find yourself.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:1}
Two fair six-sided dice are rolled, and you keep the larger number shown. On average, what do you keep?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:2}
You roll a fair die until a six appears, and you add up every number you rolled, including that final six. What
is the expected total?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, DS} \iqfirm{market maker, hedge fund, crypto} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:3}
You shuffle a standard 52-card deck and lay the cards out in a row. On average, how many pairs of neighbouring
cards have the same rank, such as two kings side by side?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:4}
Two 52-card decks are shuffled separately. You turn over one card from each deck at the same time, and carry on
through both decks. A match is a moment when the two cards are identical in rank and suit. Find the mean and variance of the number of matches.
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst, Dev} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:5}
You roll a fair die until every one of the six faces has appeared at least once. What is the expected number of
rolls?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, DS} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:6}
You toss a fair coin 21 times. A run is a longest possible block of equal results, so the tosses HHTHHH contain
three runs. What is the expected number of runs in your 21 tosses, and what is its variance?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:7}
You turn over a shuffled 52-card deck one card at a time. On average, at which position does the second ace
appear?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:8}
Three fair dice, each with faces numbered 1 to 100, are rolled. What is the exact expected value of the lowest of
the three numbers?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, QD, Dev} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:9}
You roll a fair die until some number comes up for the second time. What is the expected number of rolls, as an
exact fraction? Then: keys are placed into a hash table of 1\,024 buckets, each key going to a uniformly random
bucket. About how many keys go in before the first collision?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:10}
You draw independent random numbers, each uniform between 0 and 1, until their total exceeds 1. On average, how
many numbers do you draw?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risk, DS} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:expectation-and-indicators:11}
You draw one number from a continuous distribution and write it down. You then keep drawing independent numbers
from the same distribution until one is larger than the number you wrote down. How many further draws do you
need on average? What is the chance that one further draw is enough?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:12}
100 distinct numbers are written in a random order and read from left to right. A number is a record if it is
larger than everything before it, and the first number counts as a record. Find the mean and variance of
the number of records.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:expectation-and-indicators:13}
You roll a fair die 60 times. Let $A$ be the number of ones and $B$ the number of twos. What are the covariance
and the correlation of $A$ and $B$?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{market maker, hedge fund, crypto} \iqlevel{junior, experienced}]\label{iq:iv1:expectation-and-indicators:14}
You roll a fair die until a six appears. Someone tells you that every roll along the way was even, that is 2, 4
or 6. Given this, what is the expected number of rolls?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{market maker, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:expectation-and-indicators:15}
A pot holds 100 cooked noodles, which makes 200 loose ends. You pick two loose ends at random and tie them
together, and you repeat until no loose ends remain. Some of the tied noodles have closed into loops. What is the
expected number of loops?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader, QD} \iqfirm{market maker, crypto, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:expectation-and-indicators:16}
100 ants stand at independent uniformly random positions on a stick one metre long. Each faces left or right with
equal probability, and all walk at one metre per minute. When two ants meet, both turn around instantly. An ant
that reaches either end falls off. On average, how long is it until the last ant falls? What is the expected number
of collisions?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{bank, market maker, hedge fund} \iqlevel{junior, experienced}]\label{iq:iv1:expectation-and-indicators:17}
You turn over a shuffled 52-card deck one card at a time, and stop once you have seen at least one ace and at
least one king. On average, how many cards have you turned over?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader, DS} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:expectation-and-indicators:18}
A drawer holds 5 pairs of socks, every pair different, so 10 socks in all. You pull socks out at random, one at a
time, until you hold a matching pair. What is the expected number of socks you pull?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:expectation-and-indicators:19}
An urn holds 6 red and 4 blue balls. You draw balls at random without replacement and stop as soon as one colour
has run out. On average, how many balls are left in the urn when you stop?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, DS, Risk, QD, Dev} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:expectation-and-indicators:20}
You throw 100 balls independently into 100 bins, each ball choosing its bin uniformly at random. Find the mean and variance of the number of empty bins.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QD, Dev, LLE} \iqfirm{any} \iqlevel{junior, experienced}]\label{iq:iv1:expectation-and-indicators:21}
Quicksort is given 10 distinct numbers. It picks a pivot uniformly at random from the current list, compares every
other number in that list with the pivot, splits the list into the smaller and the larger numbers, and sorts each
part the same way. What is the expected number of comparisons? What is the formula for $n$ numbers?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:expectation-and-indicators:22}
Ten cards numbered 1 to 10 are dealt at random onto ten squares numbered 1 to 10, one card per square. Every card
that lands on its own number is removed, along with its square. The remaining cards are shuffled and dealt again
onto the remaining squares, and this repeats until no card is left. What is the expected number of deals?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{market maker, hedge fund} \iqlevel{experienced}]\label{iq:iv1:expectation-and-indicators:23}
You draw independent random numbers, each uniform between 0 and 1, and keep going for as long as each number is
larger than the one before it. You stop at the first number that is smaller than the one before it. What is the
expected value of that final number?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{market maker, hedge fund, crypto} \iqlevel{experienced}]\label{iq:iv1:expectation-and-indicators:24}
You roll a fair die until every face has appeared at least once. When you stop, some faces have appeared exactly
once; the last new face always has. What is the expected number of faces that have appeared exactly once?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:expectation-and-indicators:25}
You roll a fair die until every face has appeared at least twice. Set up an exact calculation that an interviewer
could follow on a whiteboard, and give the expected number of rolls to two decimal places.
\end{question}
