\chapter{Timed Test 9}\label{ch:iv1:timed-test-9}

\begin{ivtrains}
Take it in one sitting, without notes or a calculator.
\end{ivtrains}

\ivtestsection{A}{6}{40}

\begin{question}\label{iq:iv1:timed-test-9:1}
Two fair coins are flipped together, round after round. A coin that has already shown heads may be set aside. What is the expected number of rounds until each coin has shown heads at least once?
\end{question}

\begin{question}\label{iq:iv1:timed-test-9:2}
Three different vertices of a cube are chosen at random. How likely are they to form an equilateral triangle?
\end{question}

\begin{question}\label{iq:iv1:timed-test-9:3}
A parameter has a prior $N(0, 1)$. You observe a single measurement of 2, with independent normal noise of variance 1. What are the posterior mean and variance? What if the noise variance is 4?
\end{question}

\begin{question}\label{iq:iv1:timed-test-9:4}
A number is chosen uniformly from the divisors of $10^6$. What is the probability that it is a multiple of $10^3$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-9:5}
$X$ and $Y$ are independent standard normals. What is the probability that $X > Y + 1$, given that $P(Z < 0.71) = 0.76$ for a standard normal $Z$? What is $\mathbb{E}[X \mid X + Y = 2]$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-9:6}
$X$ and $Y$ are independent exponentials with mean 1. What are the means of the smaller and the larger? What is the probability that $X > Y + 1$?
\end{question}

\ivtestsection{B}{6}{60}

\begin{question}\label{iq:iv1:timed-test-9:7}
$N$ is Poisson with mean 1. What is the probability that $N$ is even? Separately, customers arrive at a shop as a Poisson process at a rate of 6 per hour, and each buys with probability $\frac12$, independently. What is the probability of no sale in an hour?
\end{question}

\begin{question}\label{iq:iv1:timed-test-9:8}
Five fair dice are rolled. What is the probability of exactly two pairs, meaning two values appear twice each and a third value appears once?
\end{question}

\begin{question}\label{iq:iv1:timed-test-9:9}
Two uncorrelated assets have expected excess returns of 5\% and 10\% and volatilities of 10\% and 20\%. What portfolio maximises the Sharpe ratio, and what is that ratio?
\end{question}

\begin{question}\label{iq:iv1:timed-test-9:10}
A five-card hand is dealt from a standard 52-card deck, for which there are $2\,598\,960$ possible hands. What is the probability of exactly two pairs, meaning two cards of one rank, two of another, and one card of a third rank?
\end{question}

\begin{question}\label{iq:iv1:timed-test-9:11}
A plane has 10 seats and 10 passengers, each with an assigned seat. The first has lost the boarding card and sits at random. Each later passenger takes their own seat if it is free, and otherwise a random free seat. What is the probability that the last passenger gets their own seat? What is the probability that the ninth passenger does not?
\end{question}

\begin{question}\label{iq:iv1:timed-test-9:12}
Using Stirling's approximation $\ln n! \approx n\ln n - n + \frac12\ln(2\pi n)$, how many decimal digits does $100!$ have? You may use $\ln 10 = 2.303$. What does $(n!)^{1/n}/n$ tend to as $n$ grows?
\end{question}
