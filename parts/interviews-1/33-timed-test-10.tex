\chapter{Timed Test 10}\label{ch:iv1:timed-test-10}

\begin{ivtrains}
Take it in one sitting, without notes or a calculator.
\end{ivtrains}

\ivtestsection{A}{6}{40}

\begin{question}\label{iq:iv1:timed-test-10:1}
$X$, $Y$ and $Z$ are independent and uniform on $[0, 1]$. What is the probability that $X + Y + Z$ lies between 1 and 2?
\end{question}

\begin{question}\label{iq:iv1:timed-test-10:2}
Five different balls are thrown independently and at random into 3 bins. What is the probability that no bin is empty?
\end{question}

\begin{question}\label{iq:iv1:timed-test-10:3}
A return has mean 10 and variance 4. Bound the probability of a deviation of at least 5 from the mean using Chebyshev's inequality. What is the probability if the return is normal, given that $P(Z > 2.5) = 0.0062$?
\end{question}

\begin{question}\label{iq:iv1:timed-test-10:4}
Three dice have these faces: A is (2, 2, 4, 4, 9, 9), B is (1, 1, 6, 6, 8, 8) and C is (3, 3, 5, 5, 7, 7). Each has the same total. What are the probabilities that A beats B, that B beats C, and that C beats A in a single roll against each other?
\end{question}

\begin{question}\label{iq:iv1:timed-test-10:5}
Two bus lines arrive as independent Poisson processes, with mean gaps of 10 and 20 minutes. You arrive at a random time and take the first bus. What is the probability of taking the first line? What is your expected wait?
\end{question}

\begin{question}\label{iq:iv1:timed-test-10:6}
A year on another planet has 20 days, and birthdays are independent and uniform. What is the smallest group size for which the chance of a shared birthday exceeds $\frac12$?
\end{question}

\ivtestsection{B}{6}{60}

\begin{question}\label{iq:iv1:timed-test-10:7}
$X$ is uniform on $[0, 1]$. Given $X$, $Y$ is uniform on $[0, X]$. What is $\mathbb{E}[X \mid Y = \frac12]$? You may use $\ln 2 = 0.693$.
\end{question}

\begin{question}\label{iq:iv1:timed-test-10:8}
Five fair dice are rolled. What is the probability that their total is at least 18? With six fair dice, what is the probability of a total of at least 21, given that $4\,332$ of the $46\,656$ outcomes total exactly 21?
\end{question}

\begin{question}\label{iq:iv1:timed-test-10:9}
A stream has an unknown number of items. You keep one item in memory: the first at the start, and then you replace it by the $i$-th item with probability $\frac1i$. What is the chance that the third of 5 items is the one kept at the end? How do you change this to keep 2 items, and what is the chance for each item then?
\end{question}

\begin{question}\label{iq:iv1:timed-test-10:10}
A five-card hand is dealt from a standard 52-card deck, for which there are $2\,598\,960$ possible hands. What is the probability of exactly three of a kind, meaning three cards of one rank and two other cards of two different other ranks?
\end{question}

\begin{question}\label{iq:iv1:timed-test-10:11}
In 100 flips of a fair coin, what is the probability that the number of heads differs from 50 by at least 10? Use the normal approximation with a continuity correction, given that $P(Z > 1.9) = 0.0287$.
\end{question}

\begin{question}\label{iq:iv1:timed-test-10:12}
A stock is at 100. A European call with strike 105 and one year to expiry costs 10, and the interest rate is zero. What should a European put with the same strike and expiry cost? If it is quoted at 14, what trade locks in a profit, and how much?
\end{question}
