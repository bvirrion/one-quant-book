\chapter{Continuous and Geometric Probability}\label{ch:iv1:continuous-and-geometric-probability}

\begin{ivtrains}
Turning a question about uniform or normal variables into a region whose area you can see. The early questions are pictures on a square; the later ones use symmetry on circles and spheres, the angle of a pair of normals, and the structure of spacings.
\end{ivtrains}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:1}
A stick of length 1 is cut at a uniform random point. What is the average length of the longer piece? What is the average ratio of the shorter piece to the longer one?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:2}
$X$ and $Y$ are independent and uniform on $[0, 1]$. What is the probability that $XY < 1/2$?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, DS} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:3}
$X$ and $Y$ are independent and uniform on $[0, 1]$. What are the mean and variance of $|X - Y|$?
\end{question}

\begin{question}[$\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:4}
Two people agree to meet at a cafe between noon and 1 p.m. Each arrives at an independent uniform time in that hour and waits 15 minutes for the other before leaving. What is the probability that they meet?
\end{question}

\begin{question}[$\star$ \iqroles{QR, DS, QD, QT} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:5}
A point is uniform in a unit cube in $d$ dimensions. What is the probability that it lies within $0.05$ of the boundary, for $d = 1$, 10 and 100? What is the probability that it lies in the largest ball that fits in the cube, for $d = 3$ and for $d = 10$? What does this say about high-dimensional data?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, Analyst} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:6}
A stick of length 1 is broken at two independent uniform points. What is the probability that the three pieces form a triangle? If instead you break it once at a uniform point, and then break the longer piece at a uniform point, what is it? And if you break a piece picked at random with equal chance?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:7}
$X$, $Y$ and $Z$ are independent and uniform on $[0, 1]$. What is the probability that three sticks of those lengths can form a triangle?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, QD} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:8}
A floor has parallel lines one unit apart. You drop a needle of length 1 at a random position and angle. What is the probability that it crosses a line? If the needle has length 3, how many crossings does it make on average? Find the second without doing any integral for that length.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:9}
You place $n$ points independently and uniformly on a circle. What is the probability that they all lie in some semicircle? Give the answer for $n = 3$ and $n = 5$. What does it say about whether a triangle on three points contains the centre?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:10}
You place 4 independent uniform points on a circle, join the first two with a chord and the last two with another. What is the probability that the chords cross? If 20 such points are joined in 10 fixed pairs, how many crossings are there on average?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:11}
A chord of a unit circle is chosen ``at random''. How likely is it to be longer than the side of the equilateral triangle inscribed in that circle? Compute it for three natural ways of choosing the chord, and say what the differing answers show.
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, DS} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:12}
$X$ and $Y$ are independent standard normal variables. What is the probability that $X > 0$ and $Y > \sqrt{3}\,X$? What is the probability that $X + Y > 0$, given that $X > 0$?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS, Risk} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:13}
$X$ and $Y$ are independent standard normal variables. What is the probability that $X^2 + Y^2 > 4$? What are the mean and variance of the larger of $X$ and $Y$?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Risk, DS, QD} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:14}
Three machines fail after independent exponential times with rates 1, 2 and 3 per year. What is the probability that the machine with rate 1 fails last? How long does it take on average until all three have failed?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, DS, QD, LLE} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:15}
Orders arrive as a Poisson process at 2 per minute. How likely is one order or more in a given 30 seconds? Given that exactly 3 orders came in a given minute, what is the probability that none came in its first half? When, on average, did the first of them come?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, Trader, DS} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:16}
$X$ and $Y$ are independent and uniform on $[0, 1]$. What is the distribution of the smaller of them divided by the larger? What is the average of $X/Y$, and how is that consistent with the median of $X/Y$ being 1?
\end{question}

\begin{question}[$\star\star$ \iqroles{QR, QT, QD, Dev, DS} \iqfirm{any} \iqlevel{junior}]\label{iq:iv1:continuous-and-geometric-probability:17}
You estimate $\pi$ by throwing $n$ uniform points at a unit square and counting those inside the quarter disc of radius 1 centred at one corner. What is the standard error of your estimate? About how many points do you need to be within $0.001$ with 95\% confidence?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Risk} \iqfirm{hedge fund, bank} \iqlevel{junior, experienced}]\label{iq:iv1:continuous-and-geometric-probability:18}
$X$, $Y$ and $Z$ are jointly normal with mean 0, variance 1 and pairwise correlation $0.5$. What is the probability that all three are positive? Give a formula for correlation $\rho$. What happens at $\rho = -1/2$, and why is that the lowest possible value?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:continuous-and-geometric-probability:19}
A stick of length 1 is cut at 9 independent uniform points, which gives 10 pieces. What is the average length of the shortest piece? Of the longest?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:continuous-and-geometric-probability:20}
A stick of length 1 is cut at 4 independent uniform points, which gives 5 pieces. What is the probability that the pieces can form a polygon with 5 sides? Give the formula for $k$ pieces, and say why it looks like the probability that $k$ points on a circle lie in a semicircle.
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:continuous-and-geometric-probability:21}
Three independent uniform points on a unit circle form a triangle. What is its average area?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:continuous-and-geometric-probability:22}
A point is chosen uniformly in the unit square. What is the probability that it is closer to the centre of the square than to the square's boundary?
\end{question}

\begin{question}[$\star\star\star$ \iqroles{QR, QT, Trader} \iqfirm{hedge fund, market maker} \iqlevel{junior, experienced}]\label{iq:iv1:continuous-and-geometric-probability:23}
A very large number of birds sit at independent uniform positions along a wire. Each bird paints the stretch of wire between itself and its nearest neighbour. What fraction of the wire is painted?
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:continuous-and-geometric-probability:24}
Four independent uniform points on a sphere are the corners of a tetrahedron. What is the probability that it contains the centre of the sphere? Find a proof that needs no integral.
\end{question}

\begin{question}[$\star\star\star\star$ \iqroles{QR, QT} \iqfirm{hedge fund, market maker} \iqlevel{experienced}]\label{iq:iv1:continuous-and-geometric-probability:25}
A floor has a square grid of lines, one unit apart in each direction. You drop a needle of length 1 at a random position and angle. What is the average number of grid lines it crosses? What is the probability that it crosses at least one?
\end{question}
