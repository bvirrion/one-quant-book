\section*{Chapitre \ref{ch:hf:prediction-and-sports-market-making} --- Market making de prédiction et de paris sportifs}

\begin{solution}{exo:hf:prediction-and-sports-market-making:1}
Miser sur la victoire à domicile à la cote de $1/0.47=2.13$~; miser sur l'autre issue à la cote de $1/0.55=1.82$~; les probabilités implicites totalisent 1.02~: un overround de 2~\%.
\end{solution}

\begin{solution}{exo:hf:prediction-and-sports-market-making:2}
$1\,000\times(0.70-0.47)=\$230$.
\end{solution}

\begin{solution}{exo:hf:prediction-and-sports-market-making:3}
Plus de $3-0.5=2.5$~secondes~; avec la variation des latences, environ six secondes rendent la perte négligeable.
\end{solution}

\begin{solution}{exo:hf:prediction-and-sports-market-making:4}
Le courtsider agit après environ une demi-seconde et le teneur de marché annule après environ trois~; un délai de deux secondes place l'appariement de l'ordre vers 2.5
secondes, encore avant la plupart des annulations.
\end{solution}

\begin{solution}{exo:hf:prediction-and-sports-market-making:5}
Il retarde la mise en place des ordres, mais pas leur annulation, de sorte que les teneurs de marché peuvent retirer leurs cotations périmées avant que les nouveaux ordres ne les atteignent~; retarder aussi les annulations
rendrait le dispositif symétrique et inutile.
\end{solution}

\begin{solution}{exo:hf:prediction-and-sports-market-making:6}
Savoir si les cotes des paris intègrent rapidement et pleinement l'information~; un but est une nouvelle qui tombe nettement, à un instant connu, sans fuite, et les
contrats ont des valeurs terminales. Ils ont constaté que les cotes se mettent à jour rapidement et pleinement.
\end{solution}

\begin{solution}{exo:hf:prediction-and-sports-market-making:7}
Avec un flux accusant 1.5~seconde de retard, la perte par but est de 112~\$ à une seconde, 10~\$ à deux secondes et 0.6~\$ à trois~: trois secondes la rendent négligeable au lieu de six.
\end{solution}

\begin{solution}{exo:hf:prediction-and-sports-market-making:8}
Un meilleur modèle ne protège pas contre le fait d'être plus lent que l'événement~: après un but, le courtsider en sait plus que n'importe quel modèle. Les cotations doivent être
annulées avant que les ordres fondés sur le but ne puissent s'apparier, quelle que soit la qualité du modèle.
\end{solution}

\begin{solution}{iq:hf:prediction-and-sports-market-making:1}
Environ 0.88~: en dix minutes, chaque camp marque environ 0.14 but, et le leader gagne sauf si l'autre camp en marque davantage.
\iqlookfor{Poisson sur le temps restant.}
\end{solution}

\begin{solution}{iq:hf:prediction-and-sports-market-making:2}
Retirer l'overround, puis résoudre pour les deux intensités qui reproduisent les probabilités de victoire à domicile, de match nul et à l'extérieur (et le marché du total de buts s'il existe).
\iqlookfor{inversion du modèle à partir des cotes.}
\end{solution}

\begin{solution}{iq:hf:prediction-and-sports-market-making:3}
Un bus d'événements unique alimenté par le flux le plus rapide~; par match, un message d'annulation totale précalculé pour chaque session de plateforme~; annulations de masse là où elles sont prises en charge~; surveillance de la
latence du flux.
\iqlookfor{annulation de masse sur les événements.}
\end{solution}

\begin{solution}{iq:hf:prediction-and-sports-market-making:4}
Le délai imposé par la ligue et la latence des données (un flux plus lent, un délai plus court), les nouveaux courtsiders et les événements que le modèle manque.
\iqlookfor{latence contre délai.}
\end{solution}

\begin{solution}{iq:hf:prediction-and-sports-market-making:5}
Ses paris révèlent le \omterm{def:hf:fair-value:fair}{prix juste}~; un bookmaker peut les suivre avant de limiter le compte.
\iqlookfor{information issue des comptes sharp.}
\end{solution}

\begin{solution}{iq:hf:prediction-and-sports-market-making:6}
Avec $\ln A_f\sim N(\ln a_f,s^2)$ et $\ln A_m\sim N(\ln a_m,s^2)$, $P(A_f+D<A_m)$ n'a pas de forme fermée~; conditionner sur $A_f$ et intégrer
$P(A_m>A_f+D)$ sur sa densité, numériquement.
\iqlookfor{convolution des deux latences.}
\end{solution}
