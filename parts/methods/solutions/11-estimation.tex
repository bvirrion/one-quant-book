\section*{Chapter \ref{ch:qm:estimation} --- Estimation}

\begin{solution}{exo:qm:estimation:1}
$\hat\lambda = 100/50 = 2$ per second; $\mathrm{se} = \hat\lambda/\sqrt{100} = 0.2$.
\end{solution}

\begin{solution}{exo:qm:estimation:2}
Per day $\mathrm{SR}_1 = 1/\sqrt{252} = 0.063$ and $\mathrm{se} = \sqrt{(1 + 0.002)/1\,008} = 0.0315$; annualised, $0.0315\sqrt{252} = 0.50$.
\end{solution}

\begin{solution}{exo:qm:estimation:3}
Biased: its mean is $(n - 1)\sigma^2/n$. Consistent: the bias and the variance vanish as $n \to \infty$.
\end{solution}

\begin{solution}{exo:qm:estimation:4}
The long-run variance is $\sigma^2(1 + \rho)/(1 - \rho) = 1.5\sigma^2$, so the naive \omterm{def:qm:estimation:estimator}{standard error} is $\sqrt{1.5} = 1.22$ times too small.
\end{solution}

\begin{solution}{exo:qm:estimation:5}
$\mathcal I(p) = 1/(p(1 - p))$, so $\mathrm{se} = \sqrt{0.55 \times 0.45/200} = 0.035$ and $t = 0.05/0.035 = 1.42$: not significant.
\end{solution}

\begin{solution}{exo:qm:estimation:6}
$\Var(\hat\sigma^2) \approx 2\sigma^4/n$ and $\ln\hat\sigma = \tfrac12\ln\hat\sigma^2$, so $\Var(\ln\hat\sigma) \approx \tfrac14 \cdot 2\sigma^4/(n\sigma^4) = 1/(2n)$: $\mathrm{se} =
1/\sqrt{4\,000} = 0.0158$, the Hessian value of the chapter, correct for normal data only.
\end{solution}

\begin{solution}{exo:qm:estimation:7}
62.5\% with the iid \omterm{def:qm:estimation:estimator}{standard error}, 92.2\% with Newey--West at seven lags and 93.5\% at twenty.
\end{solution}

\begin{solution}{exo:qm:estimation:8}
Consecutive trailing 20-day returns share 19 days: the series is heavily autocorrelated, so its standard deviation understates
the uncertainty of its mean, and there are only about $3 \times 252/20 = 38$ independent observations. Annualising a 20-day return
by $\sqrt{252}$ also inflates the \omterm{def:qm:estimation:sharpe}{Sharpe ratio} by $\sqrt{20}$: use non-overlapping 20-day returns annualised by $\sqrt{252/20}$, or daily P\&L,
with a HAC \omterm{def:qm:estimation:estimator}{standard error}.
\end{solution}

\begin{solution}{pb:qm:estimation:1}
\textbf{1.} Mean 4.03 bp, standard deviation 42.3 bp.
\textbf{2.} $42.3/\sqrt{1\,260} = 1.19$ bp; $t = 3.38$.
\textbf{3.} $\widehat{\mathrm{SR}} = 1.51$, iid \omterm{def:qm:estimation:estimator}{standard error} 0.45.
\textbf{4.} 0.80, 0.59, 0.37, 0.16 against 0.8, 0.6, 0.4, 0.2.
\textbf{5.} Consecutive days hold four fifths of the same positions.
\textbf{6.} $1 + 2(0.8 + 0.6 + 0.4 + 0.2) = 5$; true \omterm{def:qm:estimation:estimator}{standard error} $42\sqrt{5/1\,260} = 2.65$ bp.
\textbf{7.} $\lfloor4 \times 12.6^{2/9}\rfloor = 7$ lags: 2.32 bp and $t = 1.73$.
\textbf{8.} 2.33 bp: the Bartlett weights beyond lag four add almost nothing.
\textbf{9.} 0.87, about twice the iid value.
\textbf{10.} 62.5\% of the time.
\textbf{11.} $\hat\sigma\sqrt{5/n} = 42.3\sqrt{5/1\,260} = 2.66$ bp.
\textbf{12.} $4/2.65 = 1.51$.
\textbf{13.} $n = (3 \times 42\sqrt5/4)^2 = 4\,960$ days, about 20 years.
\textbf{14.} Sample autocovariances are biased toward zero in finite samples, and the Bartlett weights shrink the ones that
matter (lags 1 to 4) further.
\textbf{15.} Non-overlapping returns are independent, so the iid formula is right for them, at a small loss of efficiency; the
estimate of the mean is almost unchanged.
\textbf{16.} Volatility clustering (heteroskedasticity), regime changes and slow trends in the signal's edge.
\textbf{17.} The significance threshold: twenty variants tried make a $t$ of 1.7, or even 3.4, far less surprising (chapter~12).
\textbf{18.} The mean, its HAC \omterm{def:qm:estimation:estimator}{standard error} and bandwidth, the $t$-statistic, the \omterm{def:qm:estimation:sharpe}{Sharpe ratio} with its robust \omterm{def:qm:estimation:estimator}{standard error},
and the number of variants tried.
\textbf{19.} Named result: \emph{the $t$-statistic that halved}: the iid $t$-statistic of 3.38 becomes 1.73 with Newey--West

\end{solution}

\begin{solution}{iq:qm:estimation:1}
Choose the parameter that makes the observed data most probable. It is consistent, asymptotically normal and efficient (it
reaches the Cramér--Rao bound), invariant to reparametrisation, and gives \omterm{def:qm:estimation:estimator}{standard errors} from the curvature of the
log-likelihood.
\iqlookfor{the definition and the efficiency property.}
\end{solution}

\begin{solution}{iq:qm:estimation:2}
With independent daily returns its \omterm{def:qm:estimation:estimator}{standard error} is about $1/\sqrt{\text{years}} = 1$: the estimate is one \omterm{def:qm:estimation:estimator}{standard error} from
zero, a one-sided probability of about 84\% that the true value is positive under a flat prior, far from proof.
\iqlookfor{$\mathrm{se} \approx 1/\sqrt T$ for annual \omterm{def:qm:estimation:sharpe}{Sharpe ratios}.}
\end{solution}

\begin{solution}{iq:qm:estimation:3}
$A^{-1}BA^{-1}$, with $A$ the expected Hessian of the objective and $B$ the variance of its gradient. When the likelihood is
misspecified, or with least squares under heteroskedasticity, $A \ne B$ and the usual $A^{-1}$ is wrong; the sandwich is not.
\iqlookfor{bread and meat, and the information equality.}
\end{solution}

\begin{solution}{iq:qm:estimation:4}
Overlapping windows make the dependent variable autocorrelated up to the window length, so the residuals are too: the usual
\omterm{def:qm:estimation:estimator}{standard errors} are too small by roughly the square root of the overlap. Use Newey--West with at least as many lags as the
overlap (or Hansen--Hodrick), or non-overlapping observations.
\iqlookfor{overlap-induced autocorrelation and HAC.}
\end{solution}

\begin{solution}{iq:qm:estimation:5}
The variance of the score, equal to minus the expected Hessian of the log-likelihood: the curvature, the information one
observation carries about $\theta$. Cramér--Rao: an \omterm{def:qm:estimation:estimator}{unbiased estimator}'s variance is at least $\mathcal I^{-1}/n$.
\iqlookfor{curvature equals information.}
\end{solution}

\begin{solution}{iq:qm:estimation:6}
The parameter $\theta^*$ that minimises the \omterm{def:qm:estimation:kl}{Kullback--Leibler divergence} from the true distribution to the model: the best
approximation within the model. Its \omterm{def:qm:estimation:estimator}{standard errors} are the sandwich, not the inverse Hessian.
\iqlookfor{pseudo-true value and robust variance.}
\end{solution}
